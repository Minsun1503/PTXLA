	\chapter{TỔNG QUAN ĐỀ TÀI}
	
	\section{Đặt vấn đề}
	Trong lĩnh vực Thị giác máy tính (Computer Vision), bài toán xử lý phiếu thi trắc nghiệm (OMR) là một chủ đề kinh điển, thường được sử dụng để minh họa cho các kỹ thuật xử lý ảnh cơ bản như nhị phân hóa, phát hiện biên và biến đổi hình học.
	
	Tuy nhiên, khi áp dụng vào thực tế với thiết bị phổ thông như camera điện thoại, bài toán này nảy sinh nhiều thách thức kỹ thuật thú vị mà các phương pháp truyền thống khó giải quyết trọn vẹn, ví dụ như:
	\begin{enumerate}
		\item \textbf{Chất lượng ảnh đầu vào không đồng đều:} Ảnh chụp thường bị nghiêng, méo, hoặc chịu ảnh hưởng của điều kiện ánh sáng (bóng đổ, lóa sáng) khiến việc phân ngưỡng (thresholding) trở nên khó khăn.
		\item \textbf{Vấn đề nhận dạng chữ viết tay:} Ngoài việc chấm điểm trắc nghiệm, nhu cầu số hóa các thông tin viết tay (như Số báo danh) đòi hỏi sự kết hợp với các mô hình Học sâu (Deep Learning) thay vì chỉ xử lý ảnh đơn thuần.
	\end{enumerate}
	
	Với mong muốn vận dụng các kiến thức đã học trong môn Phân tích Xử lý ảnh vào một bài toán thực tế cụ thể, nhóm quyết định thực hiện đề tài \textbf{"SmartGrader"}. Đây là một nỗ lực bước đầu nhằm xây dựng chương trình hỗ trợ chấm thi quy mô nhỏ, đồng thời là cơ hội để nhóm rèn luyện kỹ năng lập trình và tư duy giải quyết vấn đề.
	
	\section{Mục tiêu đề tài}
	Mục tiêu cốt lõi của đồ án không dừng lại ở việc tạo ra một phần mềm hoàn chỉnh, mà tập trung vào việc \textbf{thử nghiệm và đánh giá hiệu quả} của các giải thuật xử lý ảnh khi áp dụng vào dữ liệu thật. Các mục tiêu cụ thể bao gồm:
	
	\begin{itemize}
		\item \textbf{Tự động hóa việc định nghĩa Template (Auto Generator):} Thay thế phương pháp đo đạc tọa độ thủ công (dễ sai sót) bằng cách lập trình sinh phiếu thi tự động. Mục tiêu là tạo ra bộ dữ liệu mẫu (Ground Truth) có độ tin cậy cao để phục vụ quá trình chấm.
		\item \textbf{Cải thiện thuật toán OMR:} Tìm hiểu và so sánh hiệu quả của kỹ thuật Phân ngưỡng thích nghi (Adaptive Thresholding) so với phân ngưỡng toàn cục trong việc xử lý các ảnh bị nhiễu hoặc ánh sáng không đều \cite{ref_omr}.
		\item \textbf{Thử nghiệm công nghệ OCR:} Bước đầu tìm hiểu cách tích hợp thư viện EasyOCR để nhận dạng các trường thông tin viết tay (Số báo danh), kết hợp với các bộ lọc dữ liệu (Regex) để đánh giá khả năng "làm sạch" kết quả của mô hình \cite{ref_ocr}.
		\item \textbf{Xây dựng quy trình xử lý (Pipeline):} Tổ chức mã nguồn thành một luồng xử lý khép kín: từ ảnh đầu vào $\rightarrow$ tiền xử lý $\rightarrow$ nhận dạng $\rightarrow$ xuất kết quả, giúp rèn luyện tư duy thiết kế hệ thống.
	\end{itemize}
	
	\section{Phạm vi và Đối tượng nghiên cứu}
	
	\subsection{Phạm vi nghiên cứu}
	Do giới hạn về thời gian và nguồn lực, đề tài tập trung giải quyết các bài toán cụ thể sau:
	\begin{itemize}
		\item \textbf{Dữ liệu đầu vào:} Chương trình chỉ xử lý các tệp tin hình ảnh (.jpg, .png) hoặc tài liệu quét (.pdf) của phiếu trả lời trắc nghiệm do chính chương trình sinh ra (theo Template định sẵn).
		\item \textbf{Công nghệ sử dụng:} Mã nguồn được viết bằng ngôn ngữ \textbf{Python}, kết hợp các thư viện mã nguồn mở phổ biến: \textbf{OpenCV} (cho xử lý ảnh), \textbf{EasyOCR} (cho nhận dạng ký tự), và \textbf{ReportLab} (để tạo phiếu thi mẫu).
		\item \textbf{Giới hạn kỹ thuật:} Chương trình yêu cầu ảnh đầu vào phải nhìn thấy rõ 4 góc định vị (neo) của phiếu thi để thực hiện phép biến đổi hình học (Warp Perspective). Đề tài chưa xử lý các trường hợp phiếu thi bị rách, mất góc hoặc biến dạng quá lớn \cite{ref_limits}.
	\end{itemize}
	
	\subsection{Đối tượng nghiên cứu}
	Đồ án tập trung tìm hiểu và vận dụng các đối tượng kiến thức sau:
	\begin{itemize}
		\item \textbf{Các thuật toán xử lý ảnh cơ bản:} Phát hiện biên (Canny Edge Detection), Tìm đường bao (Contour Retrieval) và Biến đổi phối cảnh (Perspective Transform).
		\item \textbf{Kỹ thuật nhị phân hóa:} So sánh hiệu quả giữa Otsu's Binarization và Adaptive Gaussian Thresholding trong điều kiện ánh sáng thực tế.
		\item \textbf{Công nghệ OCR:} Tìm hiểu cấu trúc và cách fine-tuning mô hình EasyOCR để nhận dạng chữ số viết tay.
	\end{itemize}
	
	\section{Phương pháp tiếp cận}
	Để mã nguồn dễ dàng quản lý và bảo trì, nhóm áp dụng tư duy tổ chức mã nguồn theo hướng \textbf{Module hóa (Modular Design)}, chia tách các thành phần xử lý riêng biệt:
	\begin{enumerate}
		\item \textbf{Module Xử lý chính (Core Processing):} Bao gồm \texttt{omr\_engine.py} (thực hiện thuật toán chấm điểm) và \texttt{ocr\_engine.py} (gọi thư viện nhận dạng ký tự). Các tham số vị trí được tách riêng ra file cấu hình \texttt{coordinates.json} thay vì viết cứng trong code (Hard-coding).
		\item \textbf{Module Hiển thị (Visualization):} File \texttt{renderer.py} chịu trách nhiệm vẽ lại kết quả chấm (vòng tròn xanh/đỏ) lên ảnh gốc để người dùng dễ quan sát.
		\item \textbf{Module Điều phối (Main Controller):} File \texttt{main.py} đóng vai trò kết nối, đọc dữ liệu ảnh đầu vào, gọi lần lượt các hàm xử lý và xuất kết quả cuối cùng ra màn hình hoặc file Excel.
	\end{enumerate}

	\chapter{CƠ SỞ LÝ THUYẾT VÀ CÔNG NGHỆ SỬ DỤNG}
	
	\section{Tổng quan về Xử lý ảnh số (Digital Image Processing)}
	Trong khuôn khổ đồ án, các kỹ thuật xử lý ảnh đóng vai trò tiền xử lý (Preprocessing) nhằm chuẩn hóa dữ liệu đầu vào. Mục tiêu là biến đổi ảnh chụp thô (raw image) thành dạng ảnh chuẩn mà máy tính có thể "hiểu" được. Các kỹ thuật chính được áp dụng bao gồm:
	
	\subsection{Ảnh xám (Grayscale)}
	Ảnh xám là ảnh mà mỗi điểm ảnh (pixel) chỉ mang thông tin về cường độ sáng (Luminance), loại bỏ hoàn toàn thông tin màu sắc (Chrominance).
	
	Việc chuyển đổi từ không gian màu RGB sang Grayscale là bước bắt buộc vì hai lý do:
	\begin{itemize}
		\item \textbf{Giảm độ phức tạp tính toán:} Giảm dữ liệu từ 3 kênh màu xuống 1 kênh, giúp tăng tốc độ xử lý lên gấp 3 lần.
		\item \textbf{Tăng hiệu quả phát hiện biên:} Các thuật toán tìm biên hoạt động dựa trên sự thay đổi cường độ sáng, việc giữ lại màu sắc thường gây nhiễu không cần thiết.
	\end{itemize}
	
	Trong mã nguồn (\texttt{src/utils/image\_utils.py}), nhóm sử dụng công thức chuyển đổi tiêu chuẩn dựa trên độ nhạy của mắt người:
	\begin{equation}
		Y = 0.299 \times R + 0.587 \times G + 0.114 \times B
	\end{equation}
	Hàm sử dụng: \texttt{cv2.cvtColor(image, cv2.COLOR\_BGR2GRAY)}.
	
	\subsection{Làm trơn ảnh (Gaussian Blur)}
	
	\textbf{Tại sao cần làm trơn?}
	Thuật toán phát hiện biên Canny (bước sau) hoạt động dựa trên đạo hàm (tính tốc độ thay đổi màu sắc). Đạo hàm cực kỳ nhạy cảm với nhiễu. Một điểm nhiễu nhỏ (ví dụ: vết bụi) cũng tạo ra sự thay đổi màu đột ngột, khiến Canny nhận nhầm đó là một cạnh. Do đó, việc làm mờ để xóa nhiễu là bắt buộc.
	
	\textbf{Cơ chế hoạt động:}
	Đồ án sử dụng bộ lọc Gaussian thay vì bộ lọc trung bình (Mean Filter) thông thường.
	\begin{itemize}
		\item \textbf{Mean Filter:} Lấy trung bình cộng tất cả các điểm lân cận. Nhược điểm là làm mờ cả các cạnh quan trọng.
		\item \textbf{Gaussian Filter:} Tính trung bình có trọng số (Weighted Average). Các điểm ảnh ở gần tâm sẽ có trọng số lớn, các điểm ở xa có trọng số nhỏ dần theo hình quả chuông (Bell Curve).
	\end{itemize}
	
	\textbf{Ma trận hạt nhân (Kernel):}
	Trong mã nguồn, hàm \texttt{cv2.GaussianBlur} sử dụng kernel kích thước $5 \times 5$. Về mặt toán học, đây là phép tích chập (convolution) ảnh gốc với một ma trận xấp xỉ Gaussian, ví dụ:
	
	\begin{equation}
		K = \frac{1}{273} 
		\begin{bmatrix} 
			1 & 4 & 7 & 4 & 1 \\ 
			4 & 16 & 26 & 16 & 4 \\ 
			7 & 26 & 41 & 26 & 7 \\ 
			4 & 16 & 26 & 16 & 4 \\ 
			1 & 4 & 7 & 4 & 1 
		\end{bmatrix}
	\end{equation}
	
	Nhìn vào ma trận trên, ta thấy điểm trung tâm (giá trị 41) đóng góp nhiều nhất vào kết quả, các điểm rìa (giá trị 1) đóng góp rất ít.
	
	\textbf{Kết quả:} Phép toán này giúp loại bỏ nhiễu hạt tần số cao (high-frequency noise) nhưng vẫn giữ lại được độ sắc nét tương đối của các đường biên chính, tạo điều kiện tốt nhất cho thuật toán Canny.
	
	\subsection{Phát hiện biên (Canny Edge Detection)}
	Thuật toán Canny được coi là chuẩn mực trong phát hiện biên nhờ khả năng tối ưu hóa giữa việc phát hiện đầy đủ các cạnh và hạn chế tối đa việc bắt nhầm nhiễu. Quy trình toán học gồm 4 bước:
	
	\textbf{Bước 1: Khử nhiễu (Noise Reduction)}
	Sử dụng bộ lọc Gaussian (như đã trình bày ở mục 2.1.2) để làm trơn ảnh, vì các toán tử đạo hàm rất nhạy cảm với nhiễu.
	
	\textbf{Bước 2: Tính Gradient cường độ (Intensity Gradient)}
	Sau khi làm trơn, thuật toán sử dụng bộ lọc Sobel theo hai hướng ngang ($G_x$) và dọc ($G_y$) để tính đạo hàm bậc nhất của ảnh.
	
	Cường độ biên (Edge Gradient) $G$ và hướng biên $\theta$ tại mỗi điểm ảnh $(x,y)$ được tính như sau:
	\begin{equation}
		G(x,y) = \sqrt{G_x^2 + G_y^2}
	\end{equation}
	\begin{equation}
		\theta(x,y) = \arctan\left(\frac{G_y}{G_x}\right)
	\end{equation}
	Hướng biên $\theta$ rất quan trọng vì nó sẽ quyết định hướng quét cho bước tiếp theo.
	
	\textbf{Bước 3: Triệt tiêu phi cực đại (Non-maximum Suppression)}
	Mục tiêu của bước này là "làm mảnh" (thinning) đường biên. Kết quả của bước 2 thường tạo ra các đường biên dày và mờ.
	
	Thuật toán sẽ duyệt qua toàn bộ các điểm ảnh. Tại mỗi điểm, nó so sánh cường độ $G$ của điểm đó với hai điểm lân cận theo hướng vuông góc với biên $\theta$.
	\begin{itemize}
		\item Nếu điểm hiện tại là lớn nhất (cực đại cục bộ): Giữ lại.
		\item Nếu không: Gán giá trị về 0 (loại bỏ).
	\end{itemize}
	Kết quả là ta thu được các đường biên mảnh (độ rộng 1 pixel).
	
	\textbf{Bước 4: Phân ngưỡng trễ (Hysteresis Thresholding)}
	Đây là bước quyết định giúp Canny vượt trội hơn các thuật toán khác trong việc xử lý biên đứt đoạn. Thuật toán phân loại các điểm ảnh thành 3 nhóm dựa trên hai ngưỡng $T_{min}$ và $T_{max}$:
	
	\begin{itemize}
		\item \textbf{Biên mạnh (Strong Edge):} Cường độ $G > T_{max}$. Điểm này chắc chắn là biên.
		\item \textbf{Biên yếu (Weak Edge):} Cường độ $T_{min} \le G \le T_{max}$. Điểm này nằm ở vùng tranh chấp.
		\item \textbf{Không phải biên (Non-Edge):} Cường độ $G < T_{min}$. Loại bỏ ngay lập tức.
	\end{itemize}
	
	\textbf{Cơ chế liên thông:} Một điểm "Biên yếu" sẽ được giữ lại (chuyển thành biên mạnh) \textbf{nếu và chỉ nếu} nó có kết nối (nằm cạnh) với một điểm "Biên mạnh". Ngược lại, nó sẽ bị coi là nhiễu và bị loại bỏ. Cơ chế này giúp nối liền các nét đứt quãng do ánh sáng yếu mà không làm tăng nhiễu.
	
	\subsection{Biến đổi phối cảnh (Perspective Transform)}
	Do góc chụp nghiêng, các đường song song trong thực tế (như cạnh của phiếu thi) sẽ hội tụ tại một điểm ảo trong ảnh, tạo ra hiện tượng méo hình thang. Để khôi phục góc nhìn trực diện (Top-down view), hệ thống sử dụng phép biến đổi phối cảnh dựa trên \textbf{Ma trận Homography}.
	
	\textbf{Cơ sở toán học:}
	Quá trình này tìm một ma trận $M$ kích thước $3 \times 3$ để ánh xạ điểm nguồn $(x, y)$ sang điểm đích $(u, v)$. Vì đây là không gian tọa độ đồng nhất (Homogeneous Coordinates), phương trình được biểu diễn như sau:
	
	\begin{equation}
		\begin{bmatrix} x' \\ y' \\ w' \end{bmatrix} = \begin{bmatrix} h_{11} & h_{12} & h_{13} \\ h_{21} & h_{22} & h_{23} \\ h_{31} & h_{32} & h_{33} \end{bmatrix} \times \begin{bmatrix} x \\ y \\ 1 \end{bmatrix}
	\end{equation}
	
	Tọa độ thực tế của điểm ảnh mới $(u, v)$ được tính bằng cách chuẩn hóa (chia cho tham số tỉ lệ $w'$):
	\begin{equation}
		u = \frac{x'}{w'} = \frac{h_{11}x + h_{12}y + h_{13}}{h_{31}x + h_{32}y + h_{33}}, \quad 
		v = \frac{y'}{w'} = \frac{h_{21}x + h_{22}y + h_{23}}{h_{31}x + h_{32}y + h_{33}}
	\end{equation}
	
	Do ma trận $M$ có 8 bậc tự do (thường chuẩn hóa $h_{33}=1$), hệ thống cần tối thiểu \textbf{4 cặp điểm tương ứng} (4 góc phiếu thi) để giải hệ phương trình tìm 8 ẩn số còn lại.
	
	\textbf{Cơ chế nội suy (Interpolation):}
	Vì tọa độ $(u, v)$ tính được từ phép biến đổi ma trận thường là số thực (ví dụ: $10.6, 20.4$), không trùng khớp với lưới pixel số nguyên của ảnh kỹ thuật số, hàm \texttt{cv2.warpPerspective} mặc định sử dụng kỹ thuật \textbf{Nội suy song tuyến tính (Bilinear Interpolation)}.
	
	
	
	\textit{Ví dụ minh họa:}
	Giả sử sau khi biến đổi, ta cần tìm giá trị màu tại điểm thực $P(10.6, 20.4)$. Điểm này nằm giữa 4 pixel nguyên:
	\begin{itemize}
		\item Góc trên-trái $A(10, 20)$
		\item Góc trên-phải $B(11, 20)$
		\item Góc dưới-trái $C(10, 21)$
		\item Góc dưới-phải $D(11, 21)$
	\end{itemize}
	
	Thay vì chỉ lấy giá trị của điểm gần nhất (Nearest Neighbor) gây ra hiện tượng răng cưa, thuật toán sẽ tính toán sự đóng góp màu sắc dựa trên khoảng cách:
	\begin{itemize}
		\item Điểm $P$ lệch về phía $x=11$ ($0.6$ đơn vị) và $y=20$ ($0.4$ đơn vị).
		\item Do đó, màu của $P$ sẽ chịu ảnh hưởng nhiều nhất từ điểm $B(11, 20)$ và ít nhất từ điểm $C(10, 21)$.
	\end{itemize}
	
	Công thức tổng quát:
	\begin{equation}
		f(x, y) \approx \sum_{i=0}^{1} \sum_{j=0}^{1} f(x_i, y_j) \cdot w_{ij}
	\end{equation}
	Trong đó $w_{ij}$ là diện tích hình chữ nhật đối diện, đảm bảo tổng trọng số bằng 1. Kỹ thuật này giúp ảnh đầu ra (output) giữ được độ mượt mà của các đường nét, đặc biệt quan trọng khi phóng to hoặc xoay ảnh.
	
	\subsection{Phân ngưỡng thích nghi (Adaptive Thresholding)}
	
	\textbf{Hạn chế của Phân ngưỡng toàn cục (Global Thresholding):}
	Các phương pháp truyền thống như Otsu tìm một giá trị ngưỡng $T$ duy nhất cho toàn bộ bức ảnh. Giả sử $T = 127$:
	\begin{itemize}
		\item Mọi điểm ảnh $< 127$ $\rightarrow$ Đen (Mực).
		\item Mọi điểm ảnh $> 127$ $\rightarrow$ Trắng (Giấy).
	\end{itemize}
	Tuy nhiên, trong thực tế khi ảnh bị bóng đổ, cường độ sáng của vùng giấy trắng trong bóng râm có thể tụt xuống mức 100. Lúc này, Otsu sẽ nhận nhầm toàn bộ vùng bóng râm đó là "Mực đen", dẫn đến kết quả sai lệch hoàn toàn.
	\begin{figure}[H]
		\centering
		\includegraphics[width=0.95\textwidth, keepaspectratio]{images/shadow_comparison_result.jpg}
		\caption{So sánh kết quả phân ngưỡng trên ảnh thực tế có bóng đổ: (1) Ảnh gốc đầu vào; (2) Kết quả thất bại khi dùng Global Threshold (Otsu); (3) Kết quả xử lý thành công bằng Adaptive Gaussian Thresholding.}
		\label{fig:shadow_comparison}
	\end{figure}
	
	\textbf{Giải pháp Adaptive Gaussian Thresholding:}
	Để khắc phục, nhóm sử dụng phương pháp phân ngưỡng thích nghi. Ngưỡng $T(x,y)$ tại mỗi điểm ảnh không cố định mà được tính toán động dựa trên các điểm lân cận:
	
	\begin{equation}
		T(x, y) = \mu_{local}(x, y) - C
	\end{equation}
	
	Trong đó:
	\begin{itemize}
		\item $\mu_{local}$: Là trung bình có trọng số Gaussian của vùng lân cận (Block Size $51 \times 51$).
		\item $C$: Hằng số trừ đi (Constant), giúp tinh chỉnh độ nhạy (được thiết lập $C=10$).
	\end{itemize}
	
	\textbf{Ví dụ minh họa cơ chế hoạt động:}
	Xét một điểm ảnh $P$ nằm trong vùng bóng râm (Shadow Region):
	\begin{itemize}
		\item Giá trị thực của điểm $P$ (giấy trắng bị tối): $I(P) = 110$.
		\item Trung bình độ sáng của vùng lân cận $P$: $\mu_{local} = 115$ (do cả vùng đều tối).
	\end{itemize}
	
	\textit{So sánh cách xử lý:}
	\begin{itemize}
		\item \textbf{Nếu dùng Global ($T=127$):} $110 < 127 \rightarrow$ Kết luận là \textbf{ĐEN} (Sai, vì thực tế đây là giấy trắng).
		\item \textbf{Nếu dùng Adaptive ($C=10$):} 
		$$ T_{local} = 115 - 10 = 105 $$
		$$ I(P) = 110 > 105 \rightarrow \text{Kết luận là \textbf{TRẮNG} (Đúng).} $$
	\end{itemize}
	
	\textbf{Kết luận:} Nhờ cơ chế "nước nổi bèo nổi" (ngưỡng thay đổi theo nền), Adaptive Thresholding tách biệt hiệu quả vết mực in ngay cả khi cường độ sáng của nền giấy thay đổi liên tục trên toàn bức ảnh.
	
	\section{Cơ sở lý thuyết về Nhận dạng quang học (Deep Learning OCR)}
	
	Để giải quyết bài toán số hóa thông tin thí sinh (Tên, Số báo danh), đồ án sử dụng thư viện EasyOCR. Khác với các công cụ OCR truyền thống (như Tesseract) dựa trên các đặc trưng thủ công, EasyOCR là một hệ thống End-to-End Deep Learning, kết hợp hai mạng nơ-ron chuyên biệt cho hai tác vụ: Phát hiện (Detection) và Nhận dạng (Recognition).
	
	\subsection{Mạng phát hiện văn bản (Text Detection): CRAFT}
	Để định vị vị trí các dòng chữ trong ảnh, EasyOCR sử dụng kiến trúc \textbf{CRAFT (Character Region Awareness for Text Detection)}.
	
	\textbf{Sự khác biệt về cách tiếp cận:}
	Hầu hết các mô hình phát hiện vật thể (như YOLO, SSD) tiếp cận theo hướng "Top-down" (dự đoán khung bao cho toàn bộ từ). Ngược lại, CRAFT tiếp cận theo hướng \textbf{"Bottom-up" (Từ dưới lên)}. Nó phát hiện từng ký tự riêng lẻ trước, sau đó mới liên kết chúng lại thành từ. Điều này giúp mô hình cực kỳ linh hoạt với các dạng văn bản có hình dạng bất kỳ.
	
	\textbf{Cơ chế Heatmap (Bản đồ nhiệt):}
	Mô hình không trả về tọa độ hộp ngay lập tức mà trả về hai bản đồ xác suất (Probability Maps) cho từng pixel:
	\begin{itemize}
		\item \textbf{Region Score:} Xác suất pixel đó là tâm của một ký tự.
		\item \textbf{Affinity Score:} Xác suất pixel đó là khoảng không gian nối giữa hai ký tự liền kề.
	\end{itemize}
	
	\textbf{Quá trình hậu xử lý (Post-processing):}
	Sau khi có được hai bản đồ nhiệt này, hệ thống thực hiện các bước:
	\begin{enumerate}
		\item \textbf{Nhị phân hóa:} Áp dụng ngưỡng (Thresholding) để lọc bỏ các vùng có xác suất thấp.
		\item \textbf{Liên kết (Linking):} Các vùng Region sẽ được nối lại với nhau nếu vùng Affinity giữa chúng có giá trị cao.
		\item \textbf{Tạo đa giác (Polygon Generation):} Thay vì vẽ hình chữ nhật cứng nhắc, CRAFT tạo ra một đường bao đa giác ôm sát theo chiều cong của dòng chữ.
	\end{enumerate}

	\subsection{Mạng nhận dạng chuỗi (Text Recognition): CRNN + CTC}
	Sau khi định vị được vùng văn bản (Text Region), bài toán chuyển sang giai đoạn nhận dạng chuỗi (Sequence Recognition). EasyOCR áp dụng kiến trúc \textbf{CRNN} (đề xuất bởi Shi et al.), bao gồm 3 thành phần chính hoạt động như một đường ống (pipeline) liền mạch:
	
	\subsubsection{1. Trích xuất đặc trưng (Feature Extraction - CNN)}
	Đầu vào là ảnh cắt (cropped image) của từ vựng đã được chuẩn hóa kích thước (thường là cao 32 pixel). 
	\begin{itemize}
		\item Nhóm sử dụng mạng nền (Backbone) là \textbf{ResNet} để trích xuất các bản đồ đặc trưng (Feature Maps).
		\item \textbf{Cơ chế Map-to-Sequence:} Đây là bước quan trọng để chuyển đổi dữ liệu từ không gian hình ảnh (2D) sang không gian chuỗi (1D). Bản đồ đặc trưng đầu ra của CNN sẽ được chia cắt theo chiều dọc (column-wise) thành các lát cắt (slices). Mỗi lát cắt này đóng vai trò là một "từ mã" (frame) đại diện cho một vùng nhỏ của ảnh theo thứ tự từ trái sang phải.
	\end{itemize}
	
	\subsubsection{2. Mô hình hóa chuỗi (Sequence Modeling - RNN)}
	Chuỗi các đặc trưng hình ảnh được đưa vào mạng \textbf{Bi-LSTM (Bidirectional Long Short-Term Memory)}.
	\begin{itemize}
		\item \textbf{Tại sao cần 2 chiều?} Chữ viết có tính ngữ cảnh mạnh. Ví dụ trong ảnh mờ, nếu chỉ nhìn từ trái sang, chữ 'H' có thể giống chữ 'N'. Nhưng nếu biết ký tự sau nó là 'o' và 't' (trong từ 'Hot'), mạng sẽ tự sửa lỗi dự đoán. Bi-LSTM quét dữ liệu theo cả hai hướng (Quá khứ $\rightarrow$ Tương lai và ngược lại) để tối ưu hóa khả năng này.
		\item Đầu ra của tầng này là xác suất dự đoán của từng ký tự tại mỗi lát cắt thời gian.
	\end{itemize}
	
	\subsubsection{3. Giải mã (Decoding - CTC Loss)}
	Đây là lớp cuối cùng giải quyết vấn đề "Bất đồng bộ" (Alignment Problem). Vì chiều rộng của ảnh (số lượng lát cắt) luôn lớn hơn số lượng ký tự thực tế (ví dụ: ảnh chữ "AI" có thể bị cắt thành 10 lát), RNN sẽ sinh ra rất nhiều ký tự trùng lặp.
	
	Thuật toán \textbf{CTC (Connectionist Temporal Classification)} giới thiệu một ký tự đặc biệt gọi là \textbf{"Blank token"} (ký hiệu: "-") để giải quyết vấn đề này qua 2 quy tắc:
	\begin{enumerate}
		\item \textbf{Gộp ký tự trùng:} Các ký tự giống nhau đứng liền kề sẽ bị gộp thành 1 (VD: $ll \rightarrow l$).
		\item \textbf{Xóa Blank:} Ký tự Blank dùng để phân cách các ký tự trùng lặp thực sự (VD: từ "Hello" có 2 chữ 'l', cần có blank ở giữa $l-l$ để không bị gộp).
	\end{enumerate}
	
	\textit{Ví dụ quá trình giải mã:}
	\begin{itemize}
		\item Đầu ra RNN (Raw): \texttt{-H-He--l-ll--lo-}
		\item Gộp trùng lặp: \texttt{-H-He-l-l-lo-}
		\item Loại bỏ Blank: \texttt{Hello}
	\end{itemize}

	\subsection{Hậu xử lý dữ liệu (Post-processing Pipeline)}
	Mô hình Deep Learning hoạt động dựa trên xác suất, do đó đầu ra thường chứa nhiễu (Stochastic Noise). Để đảm bảo tính toàn vẹn dữ liệu (Data Integrity), hệ thống áp dụng cơ chế hậu xử lý 2 bước:
	
	\subsubsection{1. Sửa lỗi nhầm lẫn (Homoglyph Correction)}
	OCR thường xuyên nhận diện sai các ký tự có hình dáng tương đồng. Trước khi lọc, hệ thống sử dụng từ điển ánh xạ (Mapping Dictionary) để khôi phục giá trị đúng dựa trên ngữ cảnh của trường thông tin:
	\begin{itemize}
		\item \textbf{Trường Số (SBD):} Ánh xạ các ký tự giống số về số.
		\\ \textit{Quy tắc:} \texttt{'O', 'o', 'D', 'Q'} $\rightarrow$ \texttt{'0'}; \texttt{'I', 'l', 'b'} $\rightarrow$ \texttt{'1'}; \texttt{'S', 's'} $\rightarrow$ \texttt{'5'}.
		\item \textbf{Trường Chữ (Họ tên):} Ngược lại, ánh xạ các số về ký tự (ví dụ: \texttt{'0'} $\rightarrow$ \texttt{'O'}).
	\end{itemize}
	
	\subsubsection{2. Lọc định dạng (Regex Filtering)}
	Sau bước sửa lỗi, Biểu thức chính quy (Regular Expression) được sử dụng như một bộ lọc tất định (Deterministic Filter) cuối cùng:
	\begin{itemize}
		\item \textbf{Trường SBD:} Pattern \texttt{\textbackslash D+} (xóa sạch các ký tự phi số còn sót lại).
		\item \textbf{Trường Họ tên:} Pattern \texttt{[\^{}a-zA-Z\textbackslash s\textbackslash p\{L\}]} (loại bỏ ký tự đặc biệt, giữ lại chữ cái bao gồm cả Tiếng Việt Unicode).
	\end{itemize}

	\section{Công nghệ và Thư viện nền tảng}
	
	\subsection{OpenCV (Xử lý ma trận ảnh)}
	OpenCV không chỉ là công cụ đọc ảnh, mà là thư viện hiện thực hóa các thuật toán Toán học thị giác máy tính. Trong đồ án, OpenCV chịu trách nhiệm thực hiện các phép biến đổi không gian (Spatial Transformations) và phân tích cấu trúc tô pô (Topological Structure Analysis) thông qua việc tìm đường bao (Contours).
	
	\subsection{NumPy (Đại số tuyến tính)}
	Trong khoa học máy tính, một bức ảnh kỹ thuật số thực chất là một ma trận đa chiều (Tensor). 
	\begin{itemize}
		\item Ảnh xám là ma trận 2D $(H \times W)$.
		\item Ảnh màu là ma trận 3D $(H \times W \times 3)$.
	\end{itemize}
	Thư viện NumPy cung cấp các cấu trúc dữ liệu hiệu năng cao để thực hiện các phép toán đại số tuyến tính (nhân ma trận, trượt cửa sổ) với tốc độ gần như C/C++, đóng vai trò nền tảng cho mọi tính toán pixel trong đồ án.
	
	\subsection{ReportLab (Sinh dữ liệu lập trình)}
	Thay vì tiếp cận theo hướng WYSIWYG (như Word), ReportLab cho phép "vẽ" file PDF thông qua các lệnh lập trình. 
	
	Ý nghĩa khoa học của việc này nằm ở tính chính xác tọa độ (Coordinate Precision). Mọi vị trí ô tròn, dòng kẻ đều được định nghĩa bằng đơn vị điểm (points) chính xác tuyệt đối, tạo ra bộ dữ liệu tham chiếu (Ground Truth) hoàn hảo để kiểm chứng độ chính xác của thuật toán chấm thi.
	
	\section{Kiến trúc Phần mềm (Software Architecture)}
	Mã nguồn không viết dồn vào một file (Monolithic) mà được tổ chức theo kiến trúc Phân lớp (Layered Architecture) để đảm bảo tính module hóa:
	
	\begin{itemize}
		\item \textbf{Data Layer (Cấu hình):} File \texttt{config.py} và \texttt{coordinates.json} tách biệt hoàn toàn dữ liệu với logic code.
		\item \textbf{Processing Layer (Model):} 
		\begin{itemize}
			\item \texttt{omr\_engine.py}: Chứa thuật toán xử lý ảnh thấp cấp (Pixel manipulation).
			\item \texttt{ocr\_engine.py}: Chứa logic gọi mô hình Deep Learning cao cấp.
		\end{itemize}
		\item \textbf{Presentation Layer (View):} File \texttt{renderer.py} chịu trách nhiệm trực quan hóa kết quả (Visualization).
		\item \textbf{Orchestration Layer (Controller):} File \texttt{main.py} điều phối luồng dữ liệu (Data Flow) giữa các lớp trên.
	\end{itemize}

			"info_fields": { 
				"name": [x, y, w, h],    // Vung chua Ten

	\section{Module Xử lý ảnh tiền kỳ (Preprocessing)}
	
	Ảnh chụp từ điện thoại thường bị biến dạng phối cảnh (nghiêng, méo). Do đó, trước khi thực hiện chấm điểm, ảnh đầu vào cần được hiệu chỉnh hình học để đưa về góc nhìn chuẩn. Quá trình này được thực hiện trong file \texttt{src/utils/image\_utils.py}.
	
	\subsection{Thuật toán phát hiện khung giấy}
	Hàm \texttt{warp\_document} áp dụng chuỗi các bước xử lý ảnh tiêu chuẩn để tách vùng giấy thi ra khỏi nền:
	
	\begin{enumerate}
		\item \textbf{Resize:} Thu nhỏ ảnh về chiều cao cố định (800px) để tối ưu hóa tốc độ xử lý mà không làm mất đi các đặc trưng hình học chính.
		\item \textbf{Grayscale \& Blur:} Chuyển ảnh sang thang độ xám và áp dụng bộ lọc Gaussian (kernel $5 \times 5$) để loại bỏ nhiễu hạt.
		\item \textbf{Canny Edge Detection:} Áp dụng thuật toán Canny để tìm các đường biên mạnh trong ảnh.
		\item \textbf{Contour Finding:} Tìm đường bao khép kín lớn nhất có 4 đỉnh, giả định đó là khung viền của tờ giấy thi.
	\end{enumerate}
	
	\subsection{Biến đổi phối cảnh (Perspective Transform)}
	Sau khi xác định được 4 góc của tờ giấy, hàm \texttt{four\_point\_transform} sẽ tính toán ma trận biến đổi để đưa ảnh về góc nhìn trực diện (Top-down view).
	
	\begin{lstlisting}[language=Python, caption={Hàm biến đổi hình học trong src/utils/image\_utils.py}]
		def four_point_transform(self, image: np.ndarray, points: np.ndarray) -> np.ndarray:
		"""
		Thuc hien Warp Perspective de nan thang anh.
		"""
		# Sap xep lai thu tu cac diem: tren-trai, tren-phai, duoi-phai, duoi-trai
		rect = self.order_points(points)
		(tl, tr, br, bl) = rect
		
		# Tinh toan chieu rong moi (lay gia tri lon nhat giua canh tren va canh duoi)
		width_a = np.sqrt(((br[0] - bl[0]) ** 2) + ((br[1] - bl[1]) ** 2))
		width_b = np.sqrt(((tr[0] - tl[0]) ** 2) + ((tr[1] - tl[1]) ** 2))
		max_width = max(int(width_a), int(width_b))
		
		# Tinh toan chieu cao moi (lay gia tri lon nhat giua canh trai va canh phai)
		height_a = np.sqrt(((tr[0] - br[0]) ** 2) + ((tr[1] - br[1]) ** 2))
		height_b = np.sqrt(((tl[0] - bl[0]) ** 2) + ((tl[1] - bl[1]) ** 2))
		max_height = max(int(height_a), int(height_b))
		
		# Dinh nghia cac diem dich (Destination points)
		dst = np.array([
		[0, 0],
		[max_width - 1, 0],
		[max_width - 1, max_height - 1],
		[0, max_height - 1]], dtype="float32")
		
		# Tinh ma tran bien doi M va ap dung len anh
		M = cv2.getPerspectiveTransform(rect, dst)
		warped = cv2.warpPerspective(image, M, (max_width, max_height))
		
		return warped
	\end{lstlisting}
	
	Kết quả của bước này là một ảnh đã được chuẩn hóa kích thước (resize về $1000 \times 1400$ pixels), loại bỏ các vùng nền thừa, sẵn sàng cho các bước xử lý tiếp theo.
	
	\section{Module Chấm thi (OMR Engine)}
	
	Module này (\texttt{src/core/omr\_engine.py}) đóng vai trò trung tâm trong việc xử lý logic chấm điểm, chịu trách nhiệm xác định chính xác các ô thí sinh đã tô trên phiếu.
	
	\subsection{Kỹ thuật Phân ngưỡng thích nghi (Adaptive Thresholding)}
	Trong quá trình thử nghiệm thực tế, nhóm nhận thấy thách thức lớn nhất nằm ở điều kiện ánh sáng không đồng nhất (ví dụ: bóng đổ do tay người chụp hoặc ánh đèn). Nếu sử dụng kỹ thuật phân ngưỡng cố định (Global Threshold), các vùng bị khuất bóng rất dễ bị nhận diện nhầm là "màu đen" (đã tô), dẫn đến sai số lớn.
	
	Để giải quyết vấn đề này, chương trình áp dụng thuật toán \textbf{Adaptive Gaussian Thresholding}. Thay vì dùng một ngưỡng chung, thuật toán này tính toán ngưỡng riêng cho từng vùng nhỏ dựa trên phân bố Gauss của các điểm lân cận.
	
	\begin{lstlisting}[language=Python, caption={Cài đặt thuật toán Adaptive Thresholding}]
		def _apply_adaptive_threshold(self, img):
		gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
		
		# Block Size = 51, C = 10 (Cac tham so thuc nghiem)
		# Tinh nguong cuc bo cho tung vung 51x51 pixels
		thresh = cv2.adaptiveThreshold(
		gray, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
		cv2.THRESH_BINARY_INV, 51, 10
		)
		return thresh
	\end{lstlisting}
	
	\subsection{Logic xác định ô tô (Bubble Detection)}
	Sau khi có được ảnh nhị phân (đen/trắng), quy trình xác định đáp án được thực hiện dựa trên mật độ điểm ảnh:
	\begin{itemize}
		\item Tại vị trí tọa độ mỗi ô trắc nghiệm (đã biết từ Template), chương trình tạo một mặt nạ (mask) hình tròn.
		\item Sử dụng hàm \texttt{cv2.countNonZero} để đếm số lượng điểm ảnh trắng (pixel đại diện cho vết mực) trong vùng mặt nạ đó.
		\item So sánh với ngưỡng \texttt{PIXEL\_THRESHOLD} (được thiết lập là 180 pixel qua thực nghiệm). Nếu vượt qua ngưỡng này, ô được coi là "Đã tô".
		\item \textbf{Cơ chế chọn lọc:} Trong trường hợp một câu hỏi có nhiều ô được tô (do tẩy xóa không sạch), ô có số lượng pixel lớn nhất sẽ được ưu tiên chọn làm đáp án cuối cùng.
	\end{itemize}
	
	\section{Module Nhận dạng chữ (OCR Engine)}
	
	Module này (\texttt{src/core/ocr\_engine.py}) tập trung vào nhiệm vụ số hóa thông tin thí sinh, cụ thể là trích xuất Số báo danh (SBD). Đây là bài toán khó do chữ viết tay của thí sinh thường đa dạng và không theo quy chuẩn.
	
	\subsection{Quy trình xử lý}
	\begin{enumerate}
		\item \textbf{Trích xuất vùng quan tâm (ROI Segmentation):} Dựa trên tọa độ vùng SBD được định nghĩa trong \texttt{coordinates.json}, chương trình cắt riêng phần ảnh chứa số báo danh để giảm nhiễu nền.
		\item \textbf{Tiền xử lý OCR:} Áp dụng \texttt{Otsu's Binarization} kết hợp với chuyển đổi ảnh xám để làm nổi bật nét bút viết tay trên nền giấy trắng.
		\item \textbf{Nhận dạng:} Gọi thư viện EasyOCR để đọc dữ liệu thô từ vùng ảnh đã xử lý.
	\end{enumerate}
	
	\subsection{Hậu xử lý dữ liệu bằng Regex}
	Kết quả thô (Raw Text) từ OCR thường chứa nhiễu do nhầm lẫn hình thái ký tự. Các lỗi phổ biến bao gồm: số "1" bị đọc thành chữ "l" hoặc "|", số "0" bị đọc thành chữ "O", hoặc xuất hiện các dấu chấm/phẩy thừa.
	
	Để chuẩn hóa, nhóm áp dụng bộ lọc Regular Expression (Regex):
	
	\begin{lstlisting}[language=Python, caption={Hàm lọc nhiễu SBD bằng Regex}]
		def _post_process_text(self, key_name: str, raw_text: str) -> str:
		text = raw_text.strip()
		
		# Chi xu ly loc so doi voi truong SBD
		if "sbd" in key_name.lower():
		# Pattern \D: Khop moi ky tu KHONG phai la so
		# Thay the toan bo ky tu rac bang chuoi rong
		clean_text = re.sub(r'\D', '', text)
		return clean_text
		
		return text
	\end{lstlisting}
	
	Kỹ thuật đơn giản này giúp loại bỏ hoàn toàn các ký tự phi số, đảm bảo Số báo danh đầu ra luôn là một chuỗi số sạch.
	
	\section{Tinh chỉnh Mô hình Nhận dạng (OCR Fine-tuning)}
	
	Trong quá trình kiểm thử, nhóm nhận thấy mô hình gốc (Pre-trained) của EasyOCR hoạt động kém hiệu quả với các mẫu chữ số viết tháu hoặc dính nét. Để cải thiện độ chính xác, nhóm đã thực hiện quy trình huấn luyện lại (Fine-tuning) mô hình trên nền tảng Kaggle.
	
	\subsection{Chuẩn bị dữ liệu (Data Preparation)}
	\begin{itemize}
		\item \textbf{Nguồn dữ liệu:} Nhóm sử dụng bộ dữ liệu mã nguồn mở \textit{OCR\_handwriting\_HAT2023} (HuggingFace) \cite{hat2023}. Bộ dữ liệu bao gồm \textbf{khoảng 100,000 mẫu ảnh} chứa các từ tiếng Việt viết tay có dấu, số và ký tự đặc biệt.
		\item \textbf{Phân chia dữ liệu:} Dữ liệu được chia theo tỷ lệ \textbf{4:1} (80\% cho huấn luyện, 20\% cho kiểm thử đánh giá).
		\item \textbf{Định dạng:} Dữ liệu ảnh và nhãn (Label) được đóng gói sang định dạng \textbf{LMDB (Lightning Memory-Mapped Database)}. Việc này giúp tối ưu hóa tốc độ đọc ghi (I/O) ngẫu nhiên, giảm thời gian chờ của GPU khi huấn luyện trên lượng dữ liệu lớn.
		\item \textbf{Tăng cường dữ liệu (Augmentation):} Để tránh hiện tượng quá khớp (Overfitting), nhóm áp dụng các kỹ thuật biến đổi ảnh ngẫu nhiên trong quá trình train: xoay nhẹ ($\pm 10^\circ$), thêm nhiễu hạt (Gaussian Noise), và thay đổi độ tương phản.
	\end{itemize}
	
	\subsection{Cấu hình Huấn luyện (Training Configuration)}
	Quá trình tinh chỉnh (Fine-tuning) được thực hiện trên môi trường \textbf{Google Colab (GPU Tesla T4)} với cấu hình chi tiết:
	\begin{itemize}
		\item \textbf{Mô hình nền (Pre-trained):} \texttt{english\_g2.pth} (Mô hình đã học tốt các nét ký tự Latin cơ bản).
		\item \textbf{Kiến trúc mạng:} ResNet (Trích xuất đặc trưng) + BiLSTM (Học chuỗi) + CTC (Dự đoán).
		\item \textbf{Siêu tham số (Hyperparameters):}
		\begin{itemize}
			\item Optimizer: \textbf{Adadelta} (Thích hợp cho mạng RNN).
			\item Learning Rate: $1.0$.
			\item Batch Size: 64.
			\item Số vòng lặp (Iterations): 3000 bước.
		\end{itemize}
	\end{itemize}
	
	\subsection{Xử lý bất đồng bộ kiến trúc (Transfer Learning Challenge)}
	Một thách thức kỹ thuật nhóm gặp phải là sự sai khác về kích thước lớp đầu ra (Output Layer) giữa mô hình gốc (97 lớp - ký tự tiếng Anh) và mô hình cần train (\textbf{197 lớp} - bao gồm tiếng Việt có dấu). Việc nạp trực tiếp trọng số sẽ gây lỗi lệch kích thước (Size Mismatch Error).
	
	\textbf{Giải pháp:} Nhóm áp dụng kỹ thuật \textit{Selective Weight Loading}. Ý tưởng là giữ lại toàn bộ trọng số của các tầng trích xuất đặc trưng (Feature Extractor) và tầng chuỗi (Sequence Modeling), chỉ loại bỏ và khởi tạo lại ngẫu nhiên lớp Linear cuối cùng (Prediction Layer).
	
	\begin{lstlisting}[language=Python, caption={Thuật toán lọc bỏ các lớp không tương thích khi nạp weight}]
		# Tai state_dict cua model pre-trained
		state_dict = torch.load(opt.saved_model)
		model_dict = model.state_dict()
		
		# Loc bo cac lop khong khop kich thuoc (cu the la lop Prediction cu)
		# Chi giu lai cac weights cua ResNet va BiLSTM
		new_state_dict = {
			k: v for k, v in state_dict.items() 
			if k in model_dict and v.shape == model_dict[k].shape
		}
		
		# Load weights da loc vao model hien tai
		# strict=False: Chap nhan viec bo qua lop Prediction (se duoc khoi tao ngau nhien)
		model.load_state_dict(new_state_dict, strict=False)
	\end{lstlisting}
	
	Giải pháp này cho phép giữ lại khả năng trích xuất đặc trưng (Feature Extraction) cực tốt của mô hình gốc, chỉ huấn luyện lại lớp dự đoán cuối cùng để phù hợp với bộ ký tự mới.

	\section{Đánh giá Thực nghiệm và Chiến lược Tối ưu hóa (Optimization Strategy)}
	
	Trong quá trình thực nghiệm trên bộ dữ liệu kiểm thử, nhóm đã ghi nhận những chênh lệch đáng kể về hiệu năng giữa các phương pháp nhận dạng. Kết quả cho thấy không có một công cụ đơn lẻ nào (Single Engine) giải quyết tốt trọn vẹn mọi trường hợp, dẫn đến nhu cầu phải áp dụng chiến lược kết hợp (Hybrid Approach).
	
	\subsection{Phân tích hạn chế của các Model OCR}
	Dựa trên kết quả log thực tế, nhóm rút ra hai nhận xét quan trọng về các mô hình Deep Learning đã thử nghiệm:
	
	\begin{itemize}
		\item \textbf{Về mô hình nhận dạng chữ (Text Recognition):} 
		Mô hình có sẵn (Pre-trained \texttt{english\_g2}) cho kết quả nhận dạng \textbf{tốt hơn} so với mô hình tự huấn luyện (Custom Fine-tuned) đối với các trường thông tin chữ in (như Tên trường, Môn thi). 
		\\ \textit{Nguyên nhân:} Mô hình Custom bị giới hạn bởi quy mô dữ liệu huấn luyện nhỏ, dẫn đến khả năng tổng quát hóa (Generalization) thấp hơn so với mô hình gốc được train trên hàng triệu dữ liệu đa dạng.
		
		\item \textbf{Về nhận dạng Số báo danh (SBD):} 
		Cả hai mô hình OCR (Gốc và Custom) đều \textbf{thất bại} trong việc nhận diện chính xác chuỗi số báo danh viết tay.
		\\ \textit{Nguyên nhân:} Ảnh cắt ra từ các ô điền số thường dính các đường kẻ khung (grid lines) hoặc bị nhiễu do nét viết chạm vào viền ô. Điều này tạo ra các đặc trưng lạ (Artifacts) khiến mạng CNN không thể trích xuất đúng đặc điểm của con số.
	\end{itemize}
	
	\subsection{Chiến lược Hợp nhất OMR - OCR}
	Thay vì cố gắng sửa lỗi cho mô hình OCR ở phần số báo danh, nhóm quyết định thay đổi chiến lược tiếp cận theo hướng "Sở trường hóa": Sử dụng OCR cho văn bản và OMR cho các ô đánh dấu.
	
	Cơ chế quyết định (Decision Logic) được thiết lập như sau:
	
	\begin{enumerate}
		\item \textbf{Đối với trường Văn bản (Tên, Lớp, Môn):} 
		Ưu tiên sử dụng \textbf{Pre-trained Model}. Mặc dù không hỗ trợ tốt dấu tiếng Việt bằng model Custom, nhưng độ ổn định và khả năng đọc chữ in của model này cao hơn, giúp định danh được các thông tin tiêu đề chính xác.
		
		\item \textbf{Đối với trường Số (SBD, Mã đề):} 
		Chúng ta \textbf{bỏ qua hoàn toàn kết quả OCR} và chỉ sử dụng kết quả từ \textbf{OMR Engine} (thuật toán xử lý ảnh xác định vị trí tô).
		\\ \textit{Lý do:} Thuật toán OMR hoạt động dựa trên mật độ điểm ảnh (Pixel Density) và vị trí hình học, do đó nó miễn nhiễm với vấn đề "nét chữ xấu" hay "dính khung" mà OCR gặp phải. OCR chỉ được kích hoạt như một kênh tham khảo phụ trong các trường hợp OMR không thể xác định (ví dụ tô mờ).
	\end{enumerate}

		Thuật toán \textbf{Adaptive Gaussian Thresholding} hoạt động xuất sắc trong điều kiện ánh sáng phức tạp (bóng đổ, chênh sáng). Hệ thống phân tách tốt vùng tô chì ngay cả khi nền giấy bị tối màu cục bộ.

		\end{itemize}
		
		\item \textbf{Độ nhạy nét tô:}
		Ngưỡng lọc (\texttt{THRESHOLD=180}) hoạt động tốt với các ô được tô kín. Tuy nhiên, hệ thống có thể bỏ sót (False Negative) đối với các trường hợp thí sinh tô quá nhạt hoặc chỉ đánh dấu gạch chéo/dấu tích mảnh.
	\end{itemize}
	
	\section{Kết quả Chấm thi (OMR Results)}

	\section{Tổng kết Hiệu năng và Hạn chế}
	
	\subsection{Hiệu năng đạt được (Performance Summary)}
	\begin{itemize}
		\item \textbf{Tốc độ xử lý (Throughput):} 
		Trung bình \textbf{1.5 - 2.0 giây/ảnh} trên cấu hình CPU phổ thông (Intel Core i5/i7, không cần GPU rời). Tốc độ này đáp ứng tốt nhu cầu chấm thi số lượng lớn trong thời gian ngắn.
		
		\item \textbf{Độ chính xác (Accuracy):} Nhờ áp dụng chiến lược Hybrid OMR-OCR, hệ thống đạt độ chính xác tối ưu cho từng loại dữ liệu:
		\begin{itemize}
			\item \textbf{Phần Trắc nghiệm \& SBD (OMR):} Đạt \textbf{> 99\%} đối với các trường hợp thí sinh tô kín ô tròn và giấy không bị nhàu nát nghiêm trọng.
			\item \textbf{Phần Tiêu đề (OCR Pre-trained):} Đạt \textbf{> 95\%} với các thông tin in sẵn (Tên trường, Môn học).
		\end{itemize}
		
		\item \textbf{Tính tự động hóa (End-to-End Automation):} 
		Hệ thống vận hành khép kín từ khâu đầu vào (ảnh chụp thô từ điện thoại) $\rightarrow$ Tiền xử lý $\rightarrow$ Nhận dạng $\rightarrow$ Xuất báo cáo Excel và ảnh chấm điểm, loại bỏ hoàn toàn thao tác nhập liệu thủ công.
	\end{itemize}
	
	\subsection{Hạn chế tồn tại (Current Limitations)}
	Mặc dù hệ thống đã đáp ứng được các yêu cầu cơ bản của bài toán chấm thi, nhóm nhìn nhận thẳng thắn các hạn chế sau:
	
	\begin{enumerate}
		\item \textbf{Nhạy cảm với biến dạng vật lý (Physical Deformation):} 
		Thuật toán \textit{Warp Perspective} giả định phiếu thi nằm trên một mặt phẳng lý tưởng. Trong thực tế, nếu ảnh chụp bị uốn cong (do cầm tay) hoặc nhăn nheo, phép biến đổi sẽ không chính xác, gây lệch vị trí các ô trắc nghiệm ở vùng trung tâm.
		
		\item \textbf{Giới hạn của OCR viết tay:} 
		Mô hình nhận dạng chữ viết tay (Handwritten Text Recognition) mặc dù đã được tinh chỉnh nhưng độ chính xác vẫn chưa đạt ngưỡng an toàn cho ứng dụng thực tế (Production-ready). Đặc biệt với trường "Họ và tên" tiếng Việt, mô hình thường xuyên nhầm lẫn dấu thanh hoặc các nét chữ tháu, đòi hỏi vẫn phải có sự xác thực lại từ con người.
		
		\item \textbf{Phụ thuộc vào mẫu cố định (Static Template):} 
		Hệ thống hoạt động dựa trên tọa độ cứng (Hard-coded coordinates) được định nghĩa trong file cấu hình. Hệ thống thiếu khả năng "tự thích nghi" (Adaptive Layout Analysis); nghĩa là chỉ cần phiếu thi thay đổi khoảng cách dòng hoặc vị trí ô, người dùng buộc phải thiết lập lại cấu hình thủ công.
	\end{enumerate}
	
	\chapter{KẾT LUẬN VÀ HƯỚNG PHÁT TRIỂN}
	
	\section{Kết luận}
	
	Đồ án \textbf{"Xây dựng chương trình hỗ trợ chấm thi và số hóa bài thi trắc nghiệm"} là bước đầu giúp nhóm tiếp cận bài toán Thị giác máy tính trong môi trường thực tế. Không chỉ dừng lại ở việc cài đặt thuật toán, đồ án còn là quá trình tư duy giải quyết vấn đề (Problem Solving) khi đối mặt với các hạn chế của dữ liệu và công nghệ.
	
	Sau quá trình nghiên cứu và thực nghiệm, nhóm rút ra các kết luận sau:
	
	\subsection{Kết quả đạt được}
	\begin{itemize}
		\item \textbf{Thử nghiệm và tích hợp mô hình lai (Hybrid OMR-OCR):} 
		Thay vì phụ thuộc vào một công nghệ duy nhất, nhóm đã triển khai song song cả hai phương pháp xử lý. Kết quả thực nghiệm giúp phân định rõ phạm vi áp dụng hiệu quả cho từng module: thuật toán \textbf{OMR} cho thấy độ ổn định cao hơn với dữ liệu số (SBD), trong khi \textbf{Deep Learning OCR} phù hợp hơn để trích xuất các thông tin văn bản (Tên trường, Môn thi).
		
		\item \textbf{Cải thiện bước Tiền xử lý (Preprocessing):}
		\begin{itemize}
			\item Kỹ thuật \textbf{Adaptive Gaussian Thresholding} đã hỗ trợ giảm thiểu tác động của hiện tượng bóng đổ (Shadow), giúp hệ thống hoạt động ổn định hơn trong điều kiện ánh sáng thường.
			\item Kỹ thuật \textbf{Perspective Transform} thực hiện tốt chức năng chuẩn hóa hình học cho các ảnh chụp từ góc nghiêng, tạo đầu vào chuẩn xác cho các bước sau.
		\end{itemize}
		
		\item \textbf{Tích hợp bước Hậu xử lý (Post-processing):} 
		Bên cạnh kết quả nhận dạng thô, hệ thống đã bổ sung các bộ lọc Regex và luật sửa lỗi (Heuristic Rules). Bước này đóng vai trò quan trọng trong việc hạn chế các lỗi nhầm lẫn ký tự điển hình của mô hình học sâu (như nhầm lẫn giữa số '0' và chữ 'O').
	\end{itemize}
	
	\subsection{Bài học kinh nghiệm}
	\begin{itemize}
		\item \textbf{Sự đánh đổi giữa Deep Learning và Thuật toán cổ điển:} 
		Qua thực nghiệm, nhóm nhận thấy mô hình học sâu (Deep Learning) dù linh hoạt nhưng thường tiêu tốn tài nguyên và có tính bất định (xác suất). Đối với các bài toán có cấu trúc ràng buộc chặt chẽ như chấm ô trắc nghiệm, các thuật toán Thị giác máy tính truyền thống (như OMR đếm pixel) lại cho thấy sự ổn định, tốc độ nhanh và khả năng kiểm soát lỗi tốt hơn. Bài học rút ra là cần lựa chọn công cụ phù hợp với đặc thù của từng bài toán con thay vì áp dụng máy móc một phương pháp.
		
		\item \textbf{Thách thức từ dữ liệu thực tế:} 
		Có một khoảng cách đáng kể giữa môi trường lý tưởng và thực tế triển khai. Các vấn đề vật lý như biến dạng giấy (cong, nhàu), điều kiện ánh sáng hay thói quen tô của thí sinh là những thách thức lớn mà thuật toán cần xử lý. Điều này khẳng định vai trò của khâu Tiền xử lý (Preprocessing) và làm sạch dữ liệu quan trọng tương đương với việc xây dựng mô hình.
	\end{itemize}
	
	\section{Hướng phát triển trong tương lai (Future Work)}
	
	Từ những hạn chế và bài học kinh nghiệm đã đúc kết, nhóm đề xuất các hướng cải tiến để nâng cấp hệ thống theo hướng thực tế hóa nếu như có thêm thời gian và nguồn lưc:
	
	\begin{enumerate}
		\item \textbf{Khắc phục biến dạng phi tuyến (Document Dewarping):} 
		Thay vì chỉ sử dụng \textit{Perspective Transform} (chỉ hiệu quả với mặt phẳng), nhóm dự kiến nghiên cứu các mô hình Deep Learning chuyên dụng (như \textit{DocTr}) để thực hiện \textbf{làm phẳng tài liệu (Dewarping)}. Kỹ thuật này sẽ giải quyết triệt để vấn đề giấy bị cong hoặc nhăn khi chụp bằng điện thoại.
		
		\item \textbf{Tự động hóa phân tích bố cục (Adaptive Layout Analysis):} 
		Thay thế cơ chế cấu hình tọa độ tĩnh (Hard-coded) bằng các thuật toán phát hiện đối tượng (Object Detection như YOLO hoặc Contour Analysis nâng cao). Mục tiêu là giúp hệ thống tự động định vị vùng SBD và vùng câu hỏi bất kể người dùng thay đổi mẫu phiếu thi như thế nào.
		
		\item \textbf{Nâng cấp mô hình nhận dạng (Transformer-based OCR):} 
		Để cải thiện độ chính xác cho chữ viết tay tiếng Việt, nhóm sẽ thử nghiệm các kiến trúc hiện đại hơn như \textbf{Vision Transformer (ViT)} hoặc \textbf{TrOCR}, thay thế cho kiến trúc CNN-RNN hiện tại vốn gặp khó khăn với các ký tự dính nét hoặc dấu thanh phức tạp.
		
		\item \textbf{Xây dựng quy trình "Human-in-the-loop":} 
		Phát triển giao diện trực quan (GUI) tích hợp cơ chế \textbf{Xác thực (Verification)}. Hệ thống sẽ tự động khoanh vùng các trường có độ tin cậy thấp (Low confidence score) để người dùng kiểm tra và sửa lỗi thủ công nhanh chóng, đảm bảo tính chính xác tuyệt đối trước khi xuất kết quả cuối cùng.
	\end{enumerate}
