	\chapter{TỔNG QUAN ĐỀ TÀI}
	
	\section{Đặt vấn đề}
	Trong lĩnh vực Thị giác máy tính (Computer Vision), bài toán xử lý phiếu thi trắc nghiệm (OMR) là một chủ đề kinh điển, thường được sử dụng để minh họa cho các kỹ thuật xử lý ảnh cơ bản như nhị phân hóa, phát hiện biên và biến đổi hình học.
	
	Tuy nhiên, khi áp dụng vào thực tế với thiết bị phổ thông như camera điện thoại, bài toán này nảy sinh nhiều thách thức kỹ thuật thú vị mà các phương pháp truyền thống khó giải quyết trọn vẹn, ví dụ như:
	\begin{enumerate}
		\item \textbf{Chất lượng ảnh đầu vào không đồng đều:} Ảnh chụp thường bị nghiêng, méo, hoặc chịu ảnh hưởng của điều kiện ánh sáng (bóng đổ, lóa sáng) khiến việc phân ngưỡng (thresholding) trở nên khó khăn.
		\item \textbf{Vấn đề nhận dạng chữ viết tay:} Ngoài việc chấm điểm trắc nghiệm, nhu cầu số hóa các thông tin viết tay (như Số báo danh) đòi hỏi sự kết hợp với các mô hình Học sâu (Deep Learning) thay vì chỉ xử lý ảnh đơn thuần.
	\end{enumerate}
	
	Với mong muốn vận dụng các kiến thức đã học trong môn Phân tích Xử lý ảnh vào một bài toán thực tế cụ thể, nhóm quyết định thực hiện đề tài \textbf{"SmartGrader"}. Đây là một nỗ lực bước đầu nhằm xây dựng chương trình hỗ trợ chấm thi quy mô nhỏ, đồng thời là cơ hội để nhóm rèn luyện kỹ năng lập trình và tư duy giải quyết vấn đề.
	
	\section{Mục tiêu đề tài}
	Mục tiêu cốt lõi của đồ án không dừng lại ở việc tạo ra một phần mềm hoàn chỉnh, mà tập trung vào việc \textbf{thử nghiệm và đánh giá hiệu quả} của các giải thuật xử lý ảnh khi áp dụng vào dữ liệu thật. Các mục tiêu cụ thể bao gồm:
	
	\begin{itemize}
		\item \textbf{Tự động hóa việc định nghĩa Template (Auto Generator):} Thay thế phương pháp đo đạc tọa độ thủ công (dễ sai sót) bằng cách lập trình sinh phiếu thi tự động. Mục tiêu là tạo ra bộ dữ liệu mẫu (Ground Truth) có độ tin cậy cao để phục vụ quá trình chấm.
		\item \textbf{Cải thiện thuật toán OMR:} Tìm hiểu và so sánh hiệu quả của kỹ thuật Phân ngưỡng thích nghi (Adaptive Thresholding) so với phân ngưỡng toàn cục trong việc xử lý các ảnh bị nhiễu hoặc ánh sáng không đều \cite{ref_omr}.
		\item \textbf{Thử nghiệm công nghệ OCR:} Bước đầu tìm hiểu cách tích hợp thư viện EasyOCR để nhận dạng các trường thông tin viết tay (Số báo danh), kết hợp với các bộ lọc dữ liệu (Regex) để đánh giá khả năng "làm sạch" kết quả của mô hình \cite{ref_ocr}.
		\item \textbf{Xây dựng quy trình xử lý (Pipeline):} Tổ chức mã nguồn thành một luồng xử lý khép kín: từ ảnh đầu vào $\rightarrow$ tiền xử lý $\rightarrow$ nhận dạng $\rightarrow$ xuất kết quả, giúp rèn luyện tư duy thiết kế hệ thống.
	\end{itemize}
	
	\section{Phạm vi và Đối tượng nghiên cứu}
	
	\subsection{Phạm vi nghiên cứu}
	Do giới hạn về thời gian và nguồn lực, đề tài tập trung giải quyết các bài toán cụ thể sau:
	\begin{itemize}
		\item \textbf{Dữ liệu đầu vào:} Chương trình chỉ xử lý các tệp tin hình ảnh (.jpg, .png) hoặc tài liệu quét (.pdf) của phiếu trả lời trắc nghiệm do chính chương trình sinh ra (theo Template định sẵn).
		\item \textbf{Công nghệ sử dụng:} Mã nguồn được viết bằng ngôn ngữ \textbf{Python}, kết hợp các thư viện mã nguồn mở phổ biến: \textbf{OpenCV} (cho xử lý ảnh), \textbf{EasyOCR} (cho nhận dạng ký tự), và \textbf{ReportLab} (để tạo phiếu thi mẫu).
		\item \textbf{Giới hạn kỹ thuật:} Chương trình yêu cầu ảnh đầu vào phải nhìn thấy rõ 4 góc định vị (neo) của phiếu thi để thực hiện phép biến đổi hình học (Warp Perspective). Đề tài chưa xử lý các trường hợp phiếu thi bị rách, mất góc hoặc biến dạng quá lớn \cite{ref_limits}.
	\end{itemize}
	
	\subsection{Đối tượng nghiên cứu}
	Đồ án tập trung tìm hiểu và vận dụng các đối tượng kiến thức sau:
	\begin{itemize}
		\item \textbf{Các thuật toán xử lý ảnh cơ bản:} Phát hiện biên (Canny Edge Detection), Tìm đường bao (Contour Retrieval) và Biến đổi phối cảnh (Perspective Transform).
		\item \textbf{Kỹ thuật nhị phân hóa:} So sánh hiệu quả giữa Otsu's Binarization và Adaptive Gaussian Thresholding trong điều kiện ánh sáng thực tế.
		\item \textbf{Công nghệ OCR:} Tìm hiểu cấu trúc và cách fine-tuning mô hình EasyOCR để nhận dạng chữ số viết tay.
	\end{itemize}
	
	\section{Phương pháp tiếp cận}
	Để mã nguồn dễ dàng quản lý và bảo trì, nhóm áp dụng tư duy tổ chức mã nguồn theo hướng \textbf{Module hóa (Modular Design)}, chia tách các thành phần xử lý riêng biệt:
	\begin{enumerate}
		\item \textbf{Module Xử lý chính (Core Processing):} Bao gồm \texttt{omr\_engine.py} (thực hiện thuật toán chấm điểm) và \texttt{ocr\_engine.py} (gọi thư viện nhận dạng ký tự). Các tham số vị trí được tách riêng ra file cấu hình \texttt{coordinates.json} thay vì viết cứng trong code (Hard-coding).
		\item \textbf{Module Hiển thị (Visualization):} File \texttt{renderer.py} chịu trách nhiệm vẽ lại kết quả chấm (vòng tròn xanh/đỏ) lên ảnh gốc để người dùng dễ quan sát.
		\item \textbf{Module Điều phối (Main Controller):} File \texttt{main.py} đóng vai trò kết nối, đọc dữ liệu ảnh đầu vào, gọi lần lượt các hàm xử lý và xuất kết quả cuối cùng ra màn hình hoặc file Excel.
	\end{enumerate}

	\end{itemize}
	
	\chapter{CƠ SỞ LÝ THUYẾT VÀ CÔNG NGHỆ SỬ DỤNG}
	
	\section{Tổng quan về Xử lý ảnh số (Digital Image Processing)}
	Trong khuôn khổ đồ án, các kỹ thuật xử lý ảnh đóng vai trò tiền xử lý (Preprocessing) nhằm chuẩn hóa dữ liệu đầu vào. Mục tiêu là biến đổi ảnh chụp thô (raw image) thành dạng ảnh chuẩn mà máy tính có thể "hiểu" được. Các kỹ thuật chính được áp dụng bao gồm:
	
	\subsection{Ảnh xám (Grayscale)}
	Ảnh xám là ảnh mà mỗi điểm ảnh (pixel) chỉ mang thông tin về cường độ sáng (Luminance), loại bỏ hoàn toàn thông tin màu sắc (Chrominance).
	
	Việc chuyển đổi từ không gian màu RGB sang Grayscale là bước bắt buộc vì hai lý do:
	\begin{itemize}
		\item \textbf{Giảm độ phức tạp tính toán:} Giảm dữ liệu từ 3 kênh màu xuống 1 kênh, giúp tăng tốc độ xử lý lên gấp 3 lần.
		\item \textbf{Tăng hiệu quả phát hiện biên:} Các thuật toán tìm biên hoạt động dựa trên sự thay đổi cường độ sáng, việc giữ lại màu sắc thường gây nhiễu không cần thiết.
	\end{itemize}
	
	Trong mã nguồn (\texttt{src/utils/image\_utils.py}), nhóm sử dụng công thức chuyển đổi tiêu chuẩn dựa trên độ nhạy của mắt người:
	\begin{equation}
		Y = 0.299 \times R + 0.587 \times G + 0.114 \times B
	\end{equation}
	Hàm sử dụng: \texttt{cv2.cvtColor(image, cv2.COLOR\_BGR2GRAY)}.
	
	\subsection{Làm trơn ảnh (Gaussian Blur)}
	Ảnh chụp thực tế luôn tồn tại nhiễu tần số cao (high-frequency noise) như nhiễu hạt (salt-and-pepper). Nếu không xử lý, các nhiễu này sẽ bị thuật toán Canny nhận nhầm là "biên".
	
	Đồ án sử dụng bộ lọc Gaussian (Gaussian Filter) đóng vai trò là một bộ lọc thông thấp (low-pass filter). Ảnh được tích chập (convolution) với một hạt nhân (kernel) $5 \times 5$. Giá trị pixel mới là trung bình có trọng số của các pixel lân cận, trong đó các pixel gần tâm có trọng số cao hơn.
	
	\subsection{Phát hiện biên (Canny Edge Detection)}
	Thay vì dùng các toán tử đơn giản như Sobel (dễ bị đứt nét), nhóm sử dụng thuật toán Canny nhờ khả năng tối ưu hóa việc phát hiện biên thông qua quy trình 4 bước chặt chẽ:
	\begin{enumerate}
		\item \textbf{Khử nhiễu:} Sử dụng Gaussian Blur như đã trình bày ở trên.
		\item \textbf{Tính Gradient:} Xác định cường độ và hướng thay đổi của độ sáng tại mỗi điểm.
		\item \textbf{Non-maximum Suppression (Triệt tiêu phi cực đại):} Kỹ thuật này giúp làm mảnh nét biên. Thuật toán sẽ loại bỏ các điểm pixel không phải là cực đại cục bộ theo hướng gradient, đảm bảo đường biên chỉ rộng 1 pixel.
		\item \textbf{Hysteresis Thresholding (Phân ngưỡng trễ):} Sử dụng hai ngưỡng (\texttt{threshold1} và \texttt{threshold2}).
		\begin{itemize}
			\item Điểm có cường độ $> T_{cao}$: Chắc chắn là biên (Strong Edge).
			\item Điểm nằm giữa $[T_{thap}, T_{cao}]$: Chỉ được coi là biên nếu nó kết nối với một điểm "Strong Edge". Cơ chế này giúp thuật toán nối liền các đường biên bị đứt quãng do ánh sáng yếu.
		\end{itemize}
	\end{enumerate}
	
	\subsection{Biến đổi phối cảnh (Perspective Transform)}
	Để xử lý hiện tượng méo hình thang do góc chụp nghiêng, hệ thống cần thực hiện phép biến đổi phối cảnh để đưa ảnh về góc nhìn trực diện (Top-down view).
	
	Về mặt toán học, đây là quá trình tìm Ma trận Homography $M$ kích thước $3 \times 3$. Do ma trận này có 8 bậc tự do (Degrees of Freedom), hệ thống yêu cầu tối thiểu 4 cặp điểm tương ứng (4 góc phiếu thi) để giải hệ phương trình:
	\begin{equation}
		\begin{bmatrix} x' \\ y' \\ w' \end{bmatrix} = \begin{bmatrix} h_{11} & h_{12} & h_{13} \\ h_{21} & h_{22} & h_{23} \\ h_{31} & h_{32} & h_{33} \end{bmatrix} \times \begin{bmatrix} x \\ y \\ 1 \end{bmatrix}
	\end{equation}
	Module \texttt{image\_utils.py} chịu trách nhiệm tính toán ma trận này và ánh xạ lại toàn bộ pixel sang hệ tọa độ mới.
	
	\subsection{Phân ngưỡng thích nghi (Adaptive Thresholding)}
	Các phương pháp phân ngưỡng toàn cục (Global Thresholding) như Otsu giả định ánh sáng phân bố đều trên toàn ảnh. Điều này không đúng với ảnh chụp thực tế có bóng đổ.
	
	Nhóm lựa chọn Adaptive Gaussian Thresholding. Ngưỡng $T(x,y)$ tại mỗi điểm ảnh không cố định mà được tính dựa trên vùng lân cận:
	\begin{equation}
		T(x, y) = \left( \sum_{i=-k}^{k} \sum_{j=-k}^{k} w(i,j) \cdot I(x+i, y+j) \right) - C
	\end{equation}
	Trong đó, $w(i,j)$ là trọng số Gaussian (các điểm gần tâm quan trọng hơn). Việc trừ đi hằng số $C$ giúp tinh chỉnh độ nhạy, đảm bảo tách biệt được vết mực ngay cả khi nền giấy tại vùng đó bị tối hơn các vùng khác.
	
	\section{Cơ sở lý thuyết về Nhận dạng quang học (Deep Learning OCR)}
	
	Để giải quyết bài toán số hóa thông tin thí sinh (Tên, Số báo danh), đồ án sử dụng thư viện EasyOCR. Khác với các công cụ OCR truyền thống (như Tesseract) dựa trên các đặc trưng thủ công, EasyOCR là một hệ thống End-to-End Deep Learning, kết hợp hai mạng nơ-ron chuyên biệt cho hai tác vụ: Phát hiện (Detection) và Nhận dạng (Recognition).
	
	\subsection{Mạng phát hiện văn bản (Text Detection): CRAFT}
	Để định vị vị trí các dòng chữ trong ảnh, EasyOCR sử dụng kiến trúc CRAFT (Character Region Awareness for Text Detection). 
	
	Khác với các mô hình phát hiện vật thể (Object Detection) như YOLO hay SSD thường dùng khung bao hình chữ nhật (Bounding Box), CRAFT tiếp cận bài toán theo hướng phân đoạn ngữ nghĩa (Segmentation). Mô hình này dự đoán hai loại bản đồ nhiệt (heatmap) cho từng pixel:
	\begin{itemize}
		\item \textbf{Region Score:} Xác suất pixel đó nằm trong vùng của một ký tự.
		\item \textbf{Affinity Score:} Xác suất pixel đó là cầu nối (liên kết) giữa hai ký tự liền kề.
	\end{itemize}
	\textbf{Ưu điểm:} Nhờ cơ chế liên kết (Affinity), CRAFT có khả năng phát hiện cực tốt các dòng văn bản bị cong, nghiêng hoặc biến dạng hình học - điều rất phổ biến khi chụp phiếu thi bằng điện thoại.
	
	\subsection{Mạng nhận dạng chuỗi (Text Recognition): CRNN + CTC}
	Sau khi đã cắt được vùng ảnh chứa văn bản (ROI), bài toán trở thành chuyển đổi hình ảnh sang chuỗi ký tự (Image-to-Sequence). Kiến trúc CRNN (Convolutional Recurrent Neural Network) được áp dụng gồm 3 tầng xử lý nối tiếp:
	
	\begin{enumerate}
		\item \textbf{Trích xuất đặc trưng (Feature Extraction - CNN):} Sử dụng mạng ResNet (Residual Network) để trích xuất các bản đồ đặc trưng (feature maps) từ ảnh đầu vào. Tầng này giúp mô hình "nhìn" thấy các nét gạch, đường cong của chữ.
		\item \textbf{Mô hình hóa chuỗi (Sequence Modeling - RNN):} Các đặc trưng được đưa vào mạng Bi-LSTM (Bidirectional Long Short-Term Memory). Mạng này quét chuỗi đặc trưng theo cả hai chiều (trái sang phải và phải sang trái) để học ngữ cảnh. 
		\\ \textit{Ví dụ:} Việc biết ký tự trước đó là 'P' và sau đó là 'X' sẽ giúp mô hình dự đoán ký tự giữa là 'T' (trong từ PTXLA) chính xác hơn.
		\item \textbf{Giải mã (Decoding - CTC Loss):} Tầng cuối cùng sử dụng thuật toán CTC (Connectionist Temporal Classification) để giải mã đầu ra của RNN thành chuỗi ký tự cuối cùng. CTC giải quyết vấn đề bất đồng bộ giữa kích thước ảnh đầu vào (width) và độ dài chuỗi ký tự đầu ra (text length) mà không cần gióng hàng (alignment) thủ công.
	\end{enumerate}

	\subsection{Hậu xử lý dữ liệu (Regex Post-processing)}
	Mô hình Deep Learning hoạt động dựa trên xác suất, do đó đầu ra thường chứa nhiễu (Stochastic Noise). Để đảm bảo tính toàn vẹn dữ liệu (Data Integrity), hệ thống sử dụng Biểu thức chính quy (Regular Expression) như một bộ lọc tất định (Deterministic Filter).
	
	Chiến lược áp dụng cho từng trường thông tin:
	\begin{itemize}
		\item \textbf{Trường Số báo danh (Numeric Field):} Sử dụng pattern \texttt{\textbackslash D} (Non-digit) để tìm và loại bỏ tất cả các ký tự không phải là số. Điều này giúp khắc phục lỗi phổ biến khi OCR nhận diện nhầm vết bẩn thành dấu chấm câu.
		\item \textbf{Trường Họ tên (Text Field):} Sử dụng pattern \texttt{[\^{}\textbackslash w\textbackslash s]} để loại bỏ các ký tự đặc biệt, chỉ giữ lại chữ cái và khoảng trắng, đảm bảo tên thí sinh đúng định dạng văn bản.
	\end{itemize}
	
	\section{Công nghệ và Thư viện nền tảng}
	
	\subsection{OpenCV (Xử lý ma trận ảnh)}
	OpenCV không chỉ là công cụ đọc ảnh, mà là thư viện hiện thực hóa các thuật toán Toán học thị giác máy tính. Trong đồ án, OpenCV chịu trách nhiệm thực hiện các phép biến đổi không gian (Spatial Transformations) và phân tích cấu trúc tô pô (Topological Structure Analysis) thông qua việc tìm đường bao (Contours).
	
	\subsection{NumPy (Đại số tuyến tính)}
	Trong khoa học máy tính, một bức ảnh kỹ thuật số thực chất là một ma trận đa chiều (Tensor). 
	\begin{itemize}
		\item Ảnh xám là ma trận 2D $(H \times W)$.
		\item Ảnh màu là ma trận 3D $(H \times W \times 3)$.
	\end{itemize}
	Thư viện NumPy cung cấp các cấu trúc dữ liệu hiệu năng cao để thực hiện các phép toán đại số tuyến tính (nhân ma trận, trượt cửa sổ) với tốc độ gần như C/C++, đóng vai trò nền tảng cho mọi tính toán pixel trong đồ án.
	
	\subsection{ReportLab (Sinh dữ liệu lập trình)}
	Thay vì tiếp cận theo hướng WYSIWYG (như Word), ReportLab cho phép "vẽ" file PDF thông qua các lệnh lập trình. 
	
	Ý nghĩa khoa học của việc này nằm ở tính chính xác tọa độ (Coordinate Precision). Mọi vị trí ô tròn, dòng kẻ đều được định nghĩa bằng đơn vị điểm (points) chính xác tuyệt đối, tạo ra bộ dữ liệu tham chiếu (Ground Truth) hoàn hảo để kiểm chứng độ chính xác của thuật toán chấm thi.
	
	\section{Kiến trúc Phần mềm (Software Architecture)}
	Mã nguồn không viết dồn vào một file (Monolithic) mà được tổ chức theo kiến trúc Phân lớp (Layered Architecture) để đảm bảo tính module hóa:
	
	\begin{itemize}
		\item \textbf{Data Layer (Cấu hình):} File \texttt{config.py} và \texttt{coordinates.json} tách biệt hoàn toàn dữ liệu với logic code.
		\item \textbf{Processing Layer (Model):} 
		\begin{itemize}
			\item \texttt{omr\_engine.py}: Chứa thuật toán xử lý ảnh thấp cấp (Pixel manipulation).
			\item \texttt{ocr\_engine.py}: Chứa logic gọi mô hình Deep Learning cao cấp.
		\end{itemize}
		\item \textbf{Presentation Layer (View):} File \texttt{renderer.py} chịu trách nhiệm trực quan hóa kết quả (Visualization).
		\item \textbf{Orchestration Layer (Controller):} File \texttt{main.py} điều phối luồng dữ liệu (Data Flow) giữa các lớp trên.
	\end{itemize}

		\item \textbf{Output Generation:} Ngoài ra, thành phần này cũng chịu trách nhiệm xuất các file dữ liệu kết quả (như ảnh \texttt{score.png} và file Excel/JSON) để phục vụ việc lưu trữ.
	\end{itemize}
	
	\subsection{Lớp Điều phối và Cấu hình (Coordination \& Configuration)}
	\begin{itemize}
		\item \textbf{Main Controller (\texttt{main.py}):} Đây là điểm bắt đầu (Entry Point) của chương trình. Module này thực hiện quy trình tuần tự: Tải cấu hình $\rightarrow$ Đọc danh sách ảnh đầu vào $\rightarrow$ Gọi Processor xử lý $\rightarrow$ Gọi Renderer xuất kết quả.
		\item \textbf{Configuration (\texttt{config.py}):} Quản lý tập trung toàn bộ tham số hệ thống như đường dẫn thư mục, các ngưỡng nhị phân (\texttt{PIXEL\_THRESHOLD}), và kích thước ảnh chuẩn. Việc tách riêng cấu hình giúp dễ dàng tinh chỉnh thuật toán mà không cần can thiệp vào code xử lý.
	\end{itemize}

		\item \textbf{Thu thập dữ liệu (Input Acquisition):}
		\begin{itemize}
			\item Chương trình quét toàn bộ các file ảnh hợp lệ (.jpg, .png) có trong thư mục đầu vào \texttt{data/raw/batch\_input/}.
		\end{itemize}
		
		\item \textbf{Vòng lặp xử lý (Processing Loop):}
		Với mỗi ảnh đầu vào, hệ thống thực hiện tuần tự các bước:

			\item \textbf{Bước 1 - Biến đổi hình học (Warping):} Tìm 4 góc định vị của phiếu thi và "nắn" thẳng ảnh về kích thước chuẩn ($1000 \times 1400$ px).
			\item \textbf{Bước 2 - Chấm điểm OMR:} Cắt vùng ảnh chứa ô trắc nghiệm theo tọa độ trong template, áp dụng phân ngưỡng để xác định ô tô và so sánh với đáp án mẫu.
			\item \textbf{Bước 3 - Nhận dạng OCR:} Cắt vùng ảnh chứa thông tin viết tay (Số báo danh) và gửi đến EasyOCR để trích xuất dữ liệu chữ.

		\begin{itemize}
			\item In kết quả chi tiết từng câu (Đúng/Sai) ra màn hình Console để theo dõi.
			\item Vẽ kết quả chấm lên ảnh gốc và lưu thành file \texttt{\_result.jpg} vào thư mục đầu ra.
		\end{itemize}
	\end{enumerate}

	\subsection{Cấu trúc Template (\texttt{coordinates.json})}
	File JSON này đóng vai trò là "bản đồ" của phiếu thi, chứa 3 thành phần chính:
	
	\begin{lstlisting}[language=python, caption={Cau truc file JSON toa do}]
		{
			"info_fields": { 
				"name": [x, y, w, h],    // Vung chua Ten
				"class": [x, y, w, h],   // Vung chua Lop
				"sbd": [x, y, w, h]      // Vung viet tay SBD
			},
			"mssv_bubbles": [
			// Mang chua toa do (x,y) cua 10 cot SBD
			[ [x0,y0], [x1,y1], ... ], 
			...
			],
			"answer_bubbles": [
			// Mang chua toa do (x,y) cua cac cau hoi (Q1 -> Q20)
			// Moi cau hoi gom 4 toa do cho A, B, C, D
			[ [xA,yA], [xB,yB], [xC,yC], [xD,yD] ],
			...
			]
		}
	\end{lstlisting}
	
	\begin{itemize}
		\item \texttt{info\_fields}: Định nghĩa tọa độ các vùng chứa chữ viết tay để phục vụ OCR.
		\item \texttt{mssv\_bubbles}: Ma trận tọa độ tâm của các ô tô trắc nghiệm phần Số báo danh (10 cột $\times$ 10 dòng).
		\item \texttt{answer\_bubbles}: Ma trận tọa độ tâm của các ô trả lời trắc nghiệm (số câu hỏi $\times$ 4 lựa chọn).

	\begin{verbatim}
		1,A
		2,B
		...
		20,C
	\end{verbatim}
	Khi nạp vào chương trình, các ký tự này sẽ được ánh xạ tự động sang số nguyên: $A \rightarrow 0, B \rightarrow 1, C \rightarrow 2, D \rightarrow 3$ để thuận tiện cho việc so sánh với chỉ số mảng trong thuật toán chấm điểm.

	\section{Module Sinh phiếu thi tự động (Auto Sheet Generator)}
	
	Việc thiết kế phiếu thi thủ công bằng các công cụ văn phòng (Word/Excel) thường gặp khó khăn trong việc xác định tọa độ chính xác của các ô trắc nghiệm. Để khắc phục, nhóm đã xây dựng module \texttt{tools/generate\_sheet.py} nhằm sinh ra phiếu thi chuẩn định dạng PDF thông qua mã lệnh (Programmatic Generation).
	
	\begin{itemize}
		\item \textbf{Công nghệ sử dụng:} Thư viện \texttt{ReportLab} của Python.
		\item \textbf{Lợi ích:} Đảm bảo tính nhất quán giữa file PDF được in ra và dữ liệu tọa độ trong hệ thống. Vị trí của các ô tròn (bubbles) và khung định vị (anchors) được tính toán dựa trên công thức toán học, tạo ra bộ Dữ liệu tham chiếu (Ground Truth) có độ tin cậy cao, giúp giảm thiểu sai số trong quá trình so khớp đáp án sau này.
	\end{itemize}
	
	\section{Module Xử lý ảnh tiền kỳ (Preprocessing)}
	
	Ảnh chụp từ điện thoại thường bị biến dạng phối cảnh (nghiêng, méo). Do đó, trước khi thực hiện chấm điểm, ảnh đầu vào cần được hiệu chỉnh hình học để đưa về góc nhìn chuẩn. Quá trình này được thực hiện trong file \texttt{src/utils/image\_utils.py}.
	
	\subsection{Thuật toán phát hiện khung giấy}
	Hàm \texttt{warp\_document} áp dụng chuỗi các bước xử lý ảnh tiêu chuẩn để tách vùng giấy thi ra khỏi nền:
	
	\begin{enumerate}
		\item \textbf{Resize:} Thu nhỏ ảnh về chiều cao cố định (800px) để tối ưu hóa tốc độ xử lý mà không làm mất đi các đặc trưng hình học chính.
		\item \textbf{Grayscale \& Blur:} Chuyển ảnh sang thang độ xám và áp dụng bộ lọc Gaussian (kernel $5 \times 5$) để loại bỏ nhiễu hạt.
		\item \textbf{Canny Edge Detection:} Áp dụng thuật toán Canny để tìm các đường biên mạnh trong ảnh.
		\item \textbf{Contour Finding:} Tìm đường bao khép kín lớn nhất có 4 đỉnh, giả định đó là khung viền của tờ giấy thi.
	\end{enumerate}
	
	\subsection{Biến đổi phối cảnh (Perspective Transform)}
	Sau khi xác định được 4 góc của tờ giấy, hàm \texttt{four\_point\_transform} sẽ tính toán ma trận biến đổi để đưa ảnh về góc nhìn trực diện (Top-down view).
	
	\begin{lstlisting}[language=Python, caption={Hàm biến đổi hình học trong src/utils/image\_utils.py}]
		def four_point_transform(self, image: np.ndarray, points: np.ndarray) -> np.ndarray:
		"""
		Thuc hien Warp Perspective de nan thang anh.
		"""
		# Sap xep lai thu tu cac diem: tren-trai, tren-phai, duoi-phai, duoi-trai
		rect = self.order_points(points)
		(tl, tr, br, bl) = rect
		
		# Tinh toan chieu rong moi (lay gia tri lon nhat giua canh tren va canh duoi)
		width_a = np.sqrt(((br[0] - bl[0]) ** 2) + ((br[1] - bl[1]) ** 2))
		width_b = np.sqrt(((tr[0] - tl[0]) ** 2) + ((tr[1] - tl[1]) ** 2))
		max_width = max(int(width_a), int(width_b))
		
		# Tinh toan chieu cao moi (lay gia tri lon nhat giua canh trai va canh phai)
		height_a = np.sqrt(((tr[0] - br[0]) ** 2) + ((tr[1] - br[1]) ** 2))
		height_b = np.sqrt(((tl[0] - bl[0]) ** 2) + ((tl[1] - bl[1]) ** 2))
		max_height = max(int(height_a), int(height_b))
		
		# Dinh nghia cac diem dich (Destination points)
		dst = np.array([
		[0, 0],
		[max_width - 1, 0],
		[max_width - 1, max_height - 1],
		[0, max_height - 1]], dtype="float32")
		
		# Tinh ma tran bien doi M va ap dung len anh
		M = cv2.getPerspectiveTransform(rect, dst)
		warped = cv2.warpPerspective(image, M, (max_width, max_height))
		
		return warped
	\end{lstlisting}
	
	Kết quả của bước này là một ảnh đã được chuẩn hóa kích thước (resize về $1000 \times 1400$ pixels), loại bỏ các vùng nền thừa, sẵn sàng cho các bước xử lý tiếp theo.
	
	\section{Module Chấm thi (OMR Engine)}
	
	Module này (\texttt{src/core/omr\_engine.py}) đóng vai trò trung tâm trong việc xử lý logic chấm điểm, chịu trách nhiệm xác định chính xác các ô thí sinh đã tô trên phiếu.
	
	\subsection{Kỹ thuật Phân ngưỡng thích nghi (Adaptive Thresholding)}
	Trong quá trình thử nghiệm thực tế, nhóm nhận thấy thách thức lớn nhất nằm ở điều kiện ánh sáng không đồng nhất (ví dụ: bóng đổ do tay người chụp hoặc ánh đèn). Nếu sử dụng kỹ thuật phân ngưỡng cố định (Global Threshold), các vùng bị khuất bóng rất dễ bị nhận diện nhầm là "màu đen" (đã tô), dẫn đến sai số lớn.
	
	Để giải quyết vấn đề này, chương trình áp dụng thuật toán \textbf{Adaptive Gaussian Thresholding}. Thay vì dùng một ngưỡng chung, thuật toán này tính toán ngưỡng riêng cho từng vùng nhỏ dựa trên phân bố Gauss của các điểm lân cận.
	
	\begin{lstlisting}[language=Python, caption={Cài đặt thuật toán Adaptive Thresholding}]
		def _apply_adaptive_threshold(self, img):
		gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
		
		# Block Size = 51, C = 10 (Cac tham so thuc nghiem)
		# Tinh nguong cuc bo cho tung vung 51x51 pixels
		thresh = cv2.adaptiveThreshold(
		gray, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
		cv2.THRESH_BINARY_INV, 51, 10
		)
		return thresh
	\end{lstlisting}
	
	\subsection{Logic xác định ô tô (Bubble Detection)}
	Sau khi có được ảnh nhị phân (đen/trắng), quy trình xác định đáp án được thực hiện dựa trên mật độ điểm ảnh:
	\begin{itemize}
		\item Tại vị trí tọa độ mỗi ô trắc nghiệm (đã biết từ Template), chương trình tạo một mặt nạ (mask) hình tròn.
		\item Sử dụng hàm \texttt{cv2.countNonZero} để đếm số lượng điểm ảnh trắng (pixel đại diện cho vết mực) trong vùng mặt nạ đó.
		\item So sánh với ngưỡng \texttt{PIXEL\_THRESHOLD} (được thiết lập là 180 pixel qua thực nghiệm). Nếu vượt qua ngưỡng này, ô được coi là "Đã tô".
		\item \textbf{Cơ chế chọn lọc:} Trong trường hợp một câu hỏi có nhiều ô được tô (do tẩy xóa không sạch), ô có số lượng pixel lớn nhất sẽ được ưu tiên chọn làm đáp án cuối cùng.
	\end{itemize}
	
	\section{Module Nhận dạng chữ (OCR Engine)}
	
	Module này (\texttt{src/core/ocr\_engine.py}) tập trung vào nhiệm vụ số hóa thông tin thí sinh, cụ thể là trích xuất Số báo danh (SBD). Đây là bài toán khó do chữ viết tay của thí sinh thường đa dạng và không theo quy chuẩn.
	
	\subsection{Quy trình xử lý}
	\begin{enumerate}
		\item \textbf{Trích xuất vùng quan tâm (ROI Segmentation):} Dựa trên tọa độ vùng SBD được định nghĩa trong \texttt{coordinates.json}, chương trình cắt riêng phần ảnh chứa số báo danh để giảm nhiễu nền.
		\item \textbf{Tiền xử lý OCR:} Áp dụng \texttt{Otsu's Binarization} kết hợp với chuyển đổi ảnh xám để làm nổi bật nét bút viết tay trên nền giấy trắng.
		\item \textbf{Nhận dạng:} Gọi thư viện EasyOCR để đọc dữ liệu thô từ vùng ảnh đã xử lý.
	\end{enumerate}
	
	\subsection{Hậu xử lý dữ liệu bằng Regex}
	Kết quả thô (Raw Text) từ OCR thường chứa nhiễu do nhầm lẫn hình thái ký tự. Các lỗi phổ biến bao gồm: số "1" bị đọc thành chữ "l" hoặc "|", số "0" bị đọc thành chữ "O", hoặc xuất hiện các dấu chấm/phẩy thừa.
	
	Để chuẩn hóa, nhóm áp dụng bộ lọc Regular Expression (Regex):
	
	\begin{lstlisting}[language=Python, caption={Hàm lọc nhiễu SBD bằng Regex}]
		def _post_process_text(self, key_name: str, raw_text: str) -> str:
		text = raw_text.strip()
		
		# Chi xu ly loc so doi voi truong SBD
		if "sbd" in key_name.lower():
		# Pattern \D: Khop moi ky tu KHONG phai la so
		# Thay the toan bo ky tu rac bang chuoi rong
		clean_text = re.sub(r'\D', '', text)
		return clean_text
		
		return text
	\end{lstlisting}
	
	Kỹ thuật đơn giản này giúp loại bỏ hoàn toàn các ký tự phi số, đảm bảo Số báo danh đầu ra luôn là một chuỗi số sạch.
	
	\section{Tinh chỉnh Mô hình Nhận dạng (OCR Fine-tuning)}
	
	Trong quá trình kiểm thử, nhóm nhận thấy mô hình gốc (Pre-trained) của EasyOCR hoạt động kém hiệu quả với các mẫu chữ số viết tháu hoặc dính nét. Để cải thiện độ chính xác, nhóm đã thực hiện quy trình huấn luyện lại (Fine-tuning) mô hình trên nền tảng Kaggle.
	
	\subsection{Chuẩn bị dữ liệu}
	\begin{itemize}
		\item \textbf{Nguồn dữ liệu:} Nhóm sử dụng bộ dữ liệu mã nguồn mở \textit{OCR\_handwritting\_HAT2023} (HuggingFace), chứa các mẫu chữ viết tay tiếng Việt đa dạng \cite{hat2023}.
		\item \textbf{Định dạng:} Dữ liệu ảnh và nhãn (Label) được đóng gói sang định dạng \textbf{LMDB (Lightning Memory-Mapped Database)} . Việc này giúp tối ưu hóa tốc độ đọc ghi (I/O) khi huấn luyện trên lượng dữ liệu lớn.
	\end{itemize}
	
	\subsection{Cấu hình Huấn luyện}
	Quá trình huấn luyện sử dụng kiến trúc mạng CRNN tiêu chuẩn với các tham số:
	\begin{itemize}
		\item \textbf{Mô hình nền:} \texttt{english\_g2.pth} (Pre-trained trên dữ liệu tổng quát).
		\item \textbf{Kiến trúc mạng:} ResNet (Trích xuất đặc trưng) + BiLSTM (Học chuỗi) + CTC (Dự đoán).
		\item \textbf{Số vòng lặp (Iterations):} 3000 bước.
		\item \textbf{Kích thước Batch:} 64.
	\end{itemize}
	
	\subsection{Xử lý bất đồng bộ kiến trúc (Selective Weight Loading)}
	Một thách thức kỹ thuật nhóm gặp phải là sự sai khác về kích thước lớp đầu ra (Output Layer) giữa mô hình gốc (97 lớp - ký tự tiếng Anh) và mô hình cần train (197 lớp - bao gồm tiếng Việt có dấu). Điều này gây lỗi khi nạp trọng số (weights).
	
	Để giải quyết, nhóm đã xây dựng hàm Tải trọng số chọn lọc:
	
	\begin{lstlisting}[language=Python, caption={Thuật toán lọc bỏ các lớp không tương thích khi nạp weight}]
		# Doan code xu ly viec load weights khi so luong class thay doi
		state_dict = torch.load(opt.saved_model)
		model_dict = model.state_dict()
		
		# Chi giu lai cac lop (layers) co kich thuoc khop nhau
		# Loc bo lop Prediction cu (vi lech so luong class)
		new_state_dict = {
			k: v for k, v in state_dict.items() 
			if k in model_dict and v.shape == model_dict[k].shape
		}
		
		# Load cac weights tuong thich, khoi tao lai ngau nhien lop Prediction moi
		model.load_state_dict(new_state_dict, strict=False)
	\end{lstlisting}
	
	Giải pháp này cho phép giữ lại khả năng trích xuất đặc trưng (Feature Extraction) cực tốt của mô hình gốc, chỉ huấn luyện lại lớp dự đoán cuối cùng để phù hợp với bộ ký tự mới.

	\section{Cơ chế Nhận dạng Kép và Chiến lược Hợp nhất (Hybrid OCR Strategy)}
	
	Trong quá trình thực nghiệm, nhóm nhận thấy một sự phân hóa rõ rệt về hiệu năng giữa hai mô hình (Gốc và Fine-tuned) trên các loại dữ liệu khác nhau. Nhận thấy không có mô hình nào tối ưu cho mọi trường hợp, chương trình áp dụng chiến lược Hybrid Strategy để tận dụng sở trường riêng biệt của từng engine.
	
	\subsection{Phân tích hiệu năng so sánh}
	Quan sát bảng ghi nhận kết quả (Log) thực tế (Hình \ref{fig:dual_mode_log}), nhóm rút ra các nhận xét sau:
	\begin{itemize}
		\item \textbf{Đối với trường thông tin chữ in/Tiếng Anh (Tên trường, Lớp, Môn học):} Mô hình Gốc (Default) hoạt động ổn định hơn. Ví dụ: Chuỗi "HCMUS" được mô hình gốc nhận diện chính xác, trong khi mô hình Custom thường nhận diện sai thành "hmus" do bị ảnh hưởng bởi tập dữ liệu huấn luyện thiên về các âm tiết tiếng Việt và chữ viết tay.
		\item \textbf{Đối với trường số viết tay (SBD, Mã đề):} Mô hình Custom (Fine-tuned) thể hiện hiệu quả vượt trội trong việc xử lý các nét chữ viết tay tự nhiên, dính nét hoặc đứt đoạn - những trường hợp mà mô hình gốc thường bỏ sót hoặc nhận diện nhầm thành ký tự chữ cái.
	\end{itemize}
	
	\subsection{Cơ chế quyết định (Decision Logic)}
	Dựa trên các phân tích trên, chương trình sử dụng thuật toán lựa chọn kết quả dựa theo ngữ cảnh của vùng ảnh (Region of Interest):
	\begin{itemize}
		\item \textbf{Quy tắc 1:} Nếu vùng nhận dạng được định nghĩa là \textbf{Số (SBD, Mã đề)} $\rightarrow$ Ưu tiên sử dụng kết quả từ \textit{Custom Engine}.
		\item \textbf{Quy tắc 2:} Nếu vùng nhận dạng là \textbf{Văn bản (Tên, Lớp, Môn)} $\rightarrow$ Ưu tiên sử dụng kết quả từ \textit{Default Engine}.
	\end{itemize}
	
	Chiến lược này giúp hệ thống đạt độ chính xác tổng thể tốt hơn, khắc phục được hiện tượng quá khớp (Overfitting) của mô hình tự huấn luyện đối với các từ vựng tiếng Anh hoặc chữ in chuyên ngành, đồng thời vẫn đảm bảo khả năng đọc số viết tay chính xác.

	Việc áp dụng \textbf{Adaptive Gaussian Thresholding} đã cho thấy hiệu quả rõ rệt so với phương pháp phân ngưỡng truyền thống:
	\begin{itemize}
		\item \textbf{Xử lý bóng đổ:} Các vùng giấy bị tối do bóng râm được cân bằng lại, chuyển về nền trắng (giá trị 255).
		\item \textbf{Tách biệt đối tượng:} Các ô tô chì và mực in được tách biệt rõ ràng (giá trị 0) khỏi nền giấy, tạo thuận lợi cho việc đếm điểm ảnh (Pixel Counting).
	\end{itemize}

	\section{Tổng kết Hiệu năng và Hạn chế}
	
	\subsection{Hiệu năng đạt được}
	\begin{itemize}
		\item \textbf{Tốc độ xử lý:} Trung bình 1.5 - 2 giây/ảnh (trên CPU thường).
		\item \textbf{Độ chính xác OMR:} Đạt mức cao (>99%) với điều kiện thí sinh tô kín ô tròn.
		\item \textbf{Tính tự động:} Quy trình vận hành xuyên suốt từ đầu vào (ảnh thô) đến đầu ra (Excel, ảnh chấm) mà không cần can thiệp thủ công.
	\end{itemize}
	
	\subsection{Hạn chế tồn tại}
	Trong quá trình thực hiện, nhóm nhận thấy một số điểm hạn chế:
	\begin{enumerate}
		\item \textbf{Phụ thuộc khung viền:} Thuật toán Warp Perspective yêu cầu ảnh chụp phải nhìn thấy rõ 4 góc định vị. Nếu ảnh bị mất góc, quá trình tiền xử lý sẽ thất bại.
		\item \textbf{Hạn chế của OCR chữ viết tay:} Mặc dù đã Fine-tune, nhưng khả năng nhận dạng chữ cái viết tay tiếng Việt (Họ tên) vẫn chưa đủ độ tin cậy để đưa vào ứng dụng thực tế, do đó nhóm mới chỉ tập trung vào nhận dạng số (SBD).
		\item \textbf{Tính tĩnh của cấu trúc:} Chương trình hoạt động dựa trên tọa độ cố định. Nếu mẫu phiếu thi thay đổi bố cục, người dùng buộc phải sinh lại file cấu hình mới thay vì chương trình tự động thích nghi.
	\end{enumerate}
	
	\chapter{KẾT LUẬN VÀ HƯỚNG PHÁT TRIỂN}
	
	\section{Kết luận}
	
	Đồ án \textbf{"Xây dựng chương trình hỗ trợ chấm thi và số hóa bài thi trắc nghiệm"} là bước đầu giúp nhóm làm quen với bài toán Xử lý ảnh và Thị giác máy tính trong thực tế. Sau quá trình nghiên cứu, cài đặt và thử nghiệm, nhóm đã rút ra được những kết quả và bài học sau:
	
	\subsection{Kết quả đạt được}
	\begin{itemize}
		\item \textbf{Hoàn thiện quy trình cơ bản:} Đã xây dựng được một quy trình xử lý ảnh khép kín (End-to-End) từ ảnh đầu vào đến kết quả chấm điểm.
		
		\item \textbf{Áp dụng thành công các thuật toán nền tảng:}
		\begin{itemize}
			\item Kỹ thuật \textbf{Adaptive Thresholding} đã chứng minh hiệu quả thực tế trong việc xử lý ảnh bị bóng đổ, vượt trội hơn so với phân ngưỡng cố định.
			\item Kỹ thuật \textbf{Warp Perspective} giúp chuẩn hóa tốt các ảnh chụp từ góc nghiêng thông thường.
		\end{itemize}
		
		\item \textbf{Nghiên cứu về OCR:} Nhóm đã bước đầu tìm hiểu cách tích hợp và huấn luyện lại (Fine-tune) mô hình Deep Learning (EasyOCR) để giải quyết bài toán nhận dạng số viết tay, đồng thời hiểu rõ tầm quan trọng của việc làm sạch dữ liệu (Data Cleaning) bằng Regex.
	\end{itemize}
	
	\subsection{Bài học kinh nghiệm}
	Qua đồ án này, nhóm nhận thấy khoảng cách giữa lý thuyết và thực tế còn khá lớn. Các thuật toán truyền thống (Canny, FindContours) tuy đơn giản nhưng rất nhạy cảm với nhiễu môi trường. Điều này cho thấy sự cần thiết của việc nghiên cứu các phương pháp hiện đại hơn (như Deep Learning) để tăng tính bền vững (Robustness) cho hệ thống.
	
	\section{Hướng phát triển trong tương lai}
	
	Dựa trên những hạn chế đã phân tích, nhóm đề xuất một số hướng cải tiến để phát triển đồ án tốt hơn:
	
	\begin{enumerate}
		\item \textbf{Nâng cấp khả năng phát hiện vùng (Object Detection):} Thay thế thuật toán tìm biên Canny bằng các mô hình Deep Learning nhẹ như \textbf{YOLO} hoặc \textbf{SSD}. Việc này sẽ giúp chương trình xác định được phiếu thi ngay cả khi ảnh có nền phức tạp hoặc bị che khuất một phần.
		
		\item \textbf{Mở rộng Dataset OCR:} Thu thập và gán nhãn thêm dữ liệu chữ viết tay tiếng Việt để tiếp tục huấn luyện mô hình, nhằm mục tiêu nhận dạng được cả Tên và Lớp của thí sinh.
		
		\item \textbf{Xây dựng Giao diện (GUI):} Phát triển giao diện người dùng trực quan (thay vì dòng lệnh CLI) để người dùng dễ dàng thao tác, kiểm tra lại các bài thi bị chấm lỗi và chỉnh sửa thủ công nếu cần thiết.
	\end{enumerate}
