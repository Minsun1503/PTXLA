	\begin{alertblock}{1. Thách thức nhận dạng (Vấn đề)}
		\footnotesize
		Tại sao không dùng Deep Learning (OCR) cho tất cả?
		\begin{itemize}
			\setlength\itemsep{0.5em}
			\item \textbf{Nhiễu cấu trúc (Structural Noise):} 
			\\ Các dòng kẻ ô (grid lines) của phiếu trắc nghiệm làm model OCR dễ bị "ảo giác" (hallucination) - nhìn dòng kẻ ra chữ cái.
			\item \textbf{Nhầm lẫn ký tự (Character Confusion):} 
			\\ OCR thường đọc nhầm nét gạch dọc `|` thành số `1` hoặc ô vuông `[]` thành chữ `O` $\to$ Sai lệch SBD nghiêm trọng.
		\end{itemize}
	\end{alertblock}
	\begin{block}{2. Giải pháp phân loại xử lý (Proposed Approach)}
		\footnotesize
		Tách biệt luồng xử lý để tận dụng ưu điểm từng thuật toán:
		\begin{itemize}
			\setlength\itemsep{0.6em}
			\item \textbf{OMR (Pixel Counting):} 
			\\ Áp dụng cho dữ liệu \textit{Cấu trúc} (SBD, Mã đề).
			\\ $\to$ \textbf{Mục tiêu:} Đạt độ ổn định cao, kiểm soát lỗi tốt hơn so với mô hình học sâu.
			
			\item \textbf{OCR (EasyOCR):} 
			\\ Áp dụng cho dữ liệu \textit{Phi cấu trúc} (Họ tên, Lớp).
			\\ $\to$ \textbf{Mục tiêu:} Nhận dạng văn bản tự do (chấp nhận mức sai số nhỏ về chính tả).
		\end{itemize}
	\end{block}
